```lua
-- description: N3 Dark Space and Gaia: portal lookup and entry, player $10 bit $2000 recorder, $0AAC write log, exec counters by ROM offset, per-frame state rows for the portal / Gaia room runs.
-- requires: mesen_kit, iog_kit, gaia_kit, m1_kit, n1_kit
--
--   n3.portal()             first live portal slot (entry in DarkSpacePortalInit .. DarkSpacePortalClose) or nil
--   n3.execs(map)           exec counters on ROM offsets {name = offset} -> {name = n}
--   n3.watchHidden(S)       write callbacks on the player actor $1010/$1011: logs every change of bit $2000 (S.hid rows) with the PC after the store
--   n3.watchGsi(S)          write callbacks on $0AAC/$0AAD (globalScriptIndex): S.gsi rows (frame, $36, scene, pc, byte, value) and a read counter
--   n3.px()                 player actor position (x, y) and tile words
-- Callbacks run BEFORE the access; a memory callback reports the PC AFTER the accessing instruction. Offsets are decimal ROM file offsets (the ; {n} tags of extract:lt).
n3 = n3 or {}

local WR  = emu.memType.snesWorkRam
local P   = emu.memType.snesPrgRom
local CPU = emu.cpuType.snes

n3.PORTAL_LO, n3.PORTAL_HI = 579253, 579345   -- DarkSpacePortalInit .. DarkSpacePortalClose
n3.MON_LO,    n3.MON_HI    = 579347, 579495   -- DarkSpaceInteractionMonitor .. DarkSpaceMonitorStartWarp

function n3.h4(v) return string.format("%04X", v) end

function n3.slotOf(lo, hi)
  for _, s in ipairs(n1.live()) do
    local r = n1.rom(n1.ent(s))
    if r >= lo and r <= hi then return s end
  end
  return nil
end
function n3.portal() return n3.slotOf(n3.PORTAL_LO, n3.PORTAL_HI) end
function n3.monitor() return n3.slotOf(n3.MON_LO, n3.MON_HI) end

function n3.execs(map)
  local cnt = {}
  for name, off in pairs(map) do
    cnt[name] = 0
    emu.addMemoryCallback(function() cnt[name] = cnt[name] + 1 end, emu.callbackType.exec, off, off, CPU, P)
  end
  return cnt
end

local function pcOff()
  local st = emu.getState()
  return ((st["cpu.k"] & 0x3F) << 16) | (st["cpu.pc"] & 0xFFFF), st["frameCount"]
end

-- the player actor is the slot in $09AA (value $1000 in every scene seen so far); the watch covers $1010/$1011 only
function n3.watchHidden(S)
  S.hid = S.hid or {}
  S.hidN = 0
  emu.addMemoryCallback(function(a, v)
    local w = a & 0xFFFF
    if w ~= 0x1011 then return end
    local old = emu.read(w, WR)
    if ((old ~ v) & 0x20) == 0 then return end
    local pc, f = pcOff()
    S.hidN = S.hidN + 1
    if #S.hid < 40 then
      S.hid[#S.hid + 1] = string.format("f=%d n36=%d sc=%02X pc=%d $1011 %02X->%02X (bit $2000 %s) bright=%s", f, kit.r16(0x36), kit.r16(0x0644), pc, old, v,
        (v & 0x20) ~= 0 and "SET" or "CLEARED", tostring(emu.getState()["ppu.screenBrightness"]))
    end
  end, emu.callbackType.write, 0x1010, 0x1011, CPU, WR)
end

function n3.watchGsi(S)
  S.gsi = S.gsi or {}
  S.gsiN, S.gsiR = 0, 0
  emu.addMemoryCallback(function(a, v)
    local w = a & 0xFFFF
    if w ~= 0x0AAC and w ~= 0x0AAD then return end
    local pc, f = pcOff()
    S.gsiN = S.gsiN + 1
    if #S.gsi < 40 then
      S.gsi[#S.gsi + 1] = string.format("f=%d n36=%d sc=%02X pc=%d [%04X]=%02X (old %02X) src=%02X", f, kit.r16(0x36), kit.r16(0x0644), pc, w, v, emu.read(w, WR), kit.r8(0x0B12))
    end
  end, emu.callbackType.write, 0x0AAC, 0x0AAD, CPU, WR)
  emu.addMemoryCallback(function(a, v)
    local w = a & 0xFFFF
    if w ~= 0x0AAC and w ~= 0x0AAD then return end
    local pc, f = pcOff()
    S.gsiR = S.gsiR + 1
    if S.gsiRows == nil then S.gsiRows = {} end
    if #S.gsiRows < 30 then
      S.gsiRows[#S.gsiRows + 1] = string.format("f=%d sc=%02X pc=%d read [%04X]=%02X src=%02X", f, kit.r16(0x0644), pc, w, v, kit.r8(0x0B12))
    end
  end, emu.callbackType.read, 0x0AAC, 0x0AAD, CPU, WR)
end

-- contact sheet of n2.grab images (cols per row) written to screenshots/N3_<name>.png; returns the byte count
function n3.sheet(imgs, cols, name, scale)
  -- f1.encode holds at most 256 colours: quantize the grabs (drop the low bits of each channel) until the sheet fits
  local masks = { 0xFFFFFF, 0xF8F8F8, 0xF0F0F0, 0xE0E0E0, 0xC0C0C0 }
  local sh
  for _, mk in ipairs(masks) do
    local seen, n = {}, 0
    for _, im in ipairs(imgs) do for _, c in ipairs(im.p) do local q = c & mk; if not seen[q] then seen[q] = true; n = n + 1 end end end
    if n <= 255 then
      if mk ~= 0xFFFFFF then for _, im in ipairs(imgs) do for i, c in ipairs(im.p) do im.p[i] = c & mk end end end
      break
    end
  end
  local d = f1.encode(f1.sheet(imgs, cols, 1), scale or 1)
  emu.mcp.writeFile("screenshots/N3_" .. name .. ".png", d)
  return #d
end

function n3.px() return kit.r16(0x1014), kit.r16(0x1016), kit.r16(0x09A6), kit.r16(0x09A8) end
```

```lua
-- description: N3 experiment bodies: portal entry (n3r.portal), each call follows a load_state_file.
-- requires: mesen_kit, iog_kit, gaia_kit, m1_kit, n1_kit, e1_display, f1_kit, n2_kit, n3_kit
--
-- n3r.portal{ dy = 16, post = 150, sheet = {0,10,..}, out = "N3_x.txt" }
--   places the player one tile south of the first live portal (disclosed poke of the player actor position $1014/$1016 and the copy $09A2/$09A4), holds Up (physical D-pad up, raw $0800)
--   until the monitor's DarkSpaceMonitorStartWarp has run, then follows the warp until `post` frames after sceneCurrent = $E6 and the screen is at full brightness.
--   Rows: every change of (player $10 bit $2000, brightness, scene, sceneNext, joypad mask) with the frame offset from the first StartWarp.
--   Contact sheet of grabs at the offsets in o.sheet (frames after the first StartWarp), written to screenshots/N3_<name>.png.
n3r = n3r or {}

local WR  = emu.memType.snesWorkRam
local P   = emu.memType.snesPrgRom
local CPU = emu.cpuType.snes

function n3r.portal(o)
  o = o or {}
  local S = {}
  n3.watchHidden(S)
  n3.watchGsi(S)
  local X = n3.execs({ entryTest = 579360, btnCheck = 579369, startWarp = 579376, exitStart = 579562, mainInit = 579499 })
  local imgs, sheetAt = {}, {}
  for _, d in ipairs(o.sheet or {}) do sheetAt[d] = true end
  local rows = {}
  local last = nil
  local startF = nil
  m1.run(function(m)
    local L = m.log
    local s = n3.portal()
    if not s then L("no portal in scene %02X", kit.r16(0x0644)); return { lines = m.lines } end
    local px, py = kit.r16(s + 0x14), kit.r16(s + 0x16)
    L("scene=%02X portal slot=%04X pos=(%d,%d) $24=%04X (variant) $0E=%04X; player form=%d $0AAC=%04X $0B12=%04X $0AB2=%04X", kit.r16(0x0644), s, px, py, kit.r16(s + 0x24), kit.r16(s + 0x0E),
      kit.r16(0x0AD4), kit.r16(0x0AAC), kit.r16(0x0B12), kit.r16(0x0AB2))
    n1.teleport(px, py + (o.dy or 16))
    m.wait(o.settle or 150) -- the camera follows a teleported player slowly
    -- per-frame sampler runs inside the coroutine loop
    local t = -1
    local brs, lastBr = {}, nil
    local function sample()
      local br = emu.getState()["ppu.screenBrightness"]
      if br ~= lastBr then brs[#brs + 1] = t .. ":" .. tostring(br); lastBr = br end
      local key = string.format("hid=%d sc=%02X sn=%02X mask=%04X", (kit.r16(0x1010) & 0x2000) ~= 0 and 1 or 0, kit.r16(0x0644), kit.r16(0x0642), kit.r16(0x065A))
      if key ~= last then
        last = key
        rows[#rows + 1] = string.format("t=%d $36=%04X %s br=%s f10=%04X gsi=%04X src=%04X px=(%d,%d)", t, kit.r16(0x36), key, tostring(br), kit.r16(0x1010), kit.r16(0x0AAC), kit.r16(0x0B12), kit.r16(0x1014), kit.r16(0x1016))
      end
      if startF and sheetAt[t] then imgs[#imgs + 1] = n2.grab(2) end
    end
    gaia.hold({ "up" })
    local n = 0
    while X.startWarp == 0 and n < 60 do m.wait(1); n = n + 1 end
    L("Up held %d frames before StartWarp (exec %d); entryTest=%d btnCheck=%d", n, X.startWarp, X.entryTest, X.btnCheck)
    gaia.release()
    startF = m1.frame(); t = 0
    sample()
    local arrived, post = false, 0
    local guard = 0
    while guard < 1500 do
      m.wait(1); t = t + 1; guard = guard + 1
      sample()
      if kit.r16(0x0644) == 0xE6 and kit.r16(0x0642) == 0 and emu.getState()["ppu.screenBrightness"] == 15 then
        if not arrived then arrived = true; L("arrived in $E6 at t=%d $36=%04X", t, kit.r16(0x36)) end

      end

    L("end t=%d scene=%02X player=(%d,%d) f10=%04X form=%d gsi=%04X $0AB2=%04X src=%04X $0B08=%04X $0B0C=%04X $0B10=%04X", t, kit.r16(0x0644), kit.r16(0x1014), kit.r16(0x1016), kit.r16(0x1010),
      kit.r16(0x0AD4), kit.r16(0x0AAC), kit.r16(0x0AB2), kit.r16(0x0B12), kit.r16(0x0B08), kit.r16(0x0B0C), kit.r16(0x0B10))

-- ------------------------------------------------------------------------------------------------ fireflies (OQ-I056-2)
-- FireflyParticle entry range 587420..587473 (ROM offsets, ; {n} tags), FireflySpawnerLoop 587399..587418, spawn site 587411.
-- Tracks every firefly slot from its first live frame to its death: start x / y, camera H scroll ($068A) and V scroll ($068C), moveXAlt / moveYAlt ($7F0018,X / $7F001A,X = 0x10018 / 0x1001A + slot),
-- per-frame (dx, dy) runs, lifetime in frames, last y; spawn spacing from the spawner's $08 timer.
-- n3r.fireflies{frames = 1500, detail = 6}
function n3r.fireflies(o)
  o = o or {}
  local FF_LO, FF_HI = 587420, 587473
  local X = n3.execs({ spawnSite = 587411, spawnerTop = 587399, ffEntry = 587420 })
  local spawnF = {}
  emu.addMemoryCallback(function() spawnF[#spawnF + 1] = m1.t end, emu.callbackType.exec, 587411, 587411, CPU, P)
  local fl = {}   -- slot -> record
  local done = {}
  local n36ctl = 0
  m1.run(function(m)
    local L = m.log
    L("start scene=%02X $068A=%04X $068C=%04X mapBounds=(%d,%d) playerPos=(%d,%d)", kit.r16(0x0644), kit.r16(0x068A), kit.r16(0x068C), kit.r16(0x0692), kit.r16(0x0696), kit.r16(0x1014), kit.r16(0x1016))
    for t = 1, o.frames or 1500 do
      m.wait(1)
      local seen = {}
      for _, s in ipairs(n1.live()) do
        local r = n1.rom(n1.ent(s))
        if r >= FF_LO and r <= FF_HI then
          seen[s] = true
          local x, y = kit.r16(s + 0x14), kit.r16(s + 0x16)
          local rec = fl[s]
          if not rec then
            rec = { t0 = t, x0 = x, y0 = y, cx = kit.r16(0x068A), cy = kit.r16(0x068C), mx = kit.r16(0x10018 + s), my = kit.r16(0x1001A + s), f12 = kit.r16(s + 0x12), f10 = kit.r16(s + 0x10),
                    px = x, py = y, steps = {}, n = 0, cxs = {} }
            fl[s] = rec
          else
            rec.n = rec.n + 1
            rec.steps[#rec.steps + 1] = string.format("%d,%d", x - rec.px, y - rec.py)
            rec.cxs[#rec.cxs + 1] = kit.r16(0x068A)
            rec.px, rec.py = x, y
          end
        end
      end
      for s, rec in pairs(fl) do
        if not seen[s] then
          rec.t1 = t; rec.xl, rec.yl = rec.px, rec.py
          done[#done + 1] = rec
          fl[s] = nil
        end
      end
    end
    for s, rec in pairs(fl) do rec.t1 = nil; rec.xl, rec.yl = rec.px, rec.py; done[#done + 1] = rec end
    table.sort(done, function(a, b) return a.t0 < b.t0 end)
    -- run-length encode a step list
    local function rle(st)
      local out, cur, cnt = {}, nil, 0
      for _, v in ipairs(st) do
        if v == cur then cnt = cnt + 1 else if cur then out[#out + 1] = cur .. "x" .. cnt end cur, cnt = v, 1 end
      end
      if cur then out[#out + 1] = cur .. "x" .. cnt end
      return table.concat(out, " ")
    end
    local nfull, xmin, xmax, onscreen, offscreen = 0, 99999, -1, 0, 0
    local byChain = {}
    for i, r in ipairs(done) do
      if r.t1 then nfull = nfull + 1 end
      xmin = math.min(xmin, r.x0); xmax = math.max(xmax, r.x0)
      local sx = r.x0 - r.cx
      if sx >= 0 and sx < 256 then onscreen = onscreen + 1 else offscreen = offscreen + 1 end
      local key = string.format("mx=%d my=%d", r.mx, r.my)
      local b = byChain[key] or { n = 0, life = {}, dx = {}, dy = {} }
      b.n = b.n + 1
      if r.t1 then b.life[#b.life + 1] = r.t1 - r.t0 end
      b.rle = b.rle or rle(r.steps)
      byChain[key] = b
      if i <= (o.detail or 6) then
        L("ff#%d t0=%d x0=%d y0=%d (screen x0-$068A=%d, cam=(%04X,%04X)) mx=%d my=%d $10=%04X $12=%04X life=%s lastXY=(%d,%d) steps: %s", i, r.t0, r.x0, r.y0, sx, r.cx, r.cy, r.mx, r.my, r.f10, r.f12,
          r.t1 and (r.t1 - r.t0) or "alive", r.xl, r.yl, rle(r.steps))
      end
    end
    L("spawned=%d finished=%d x0 min=%d max=%d; on screen at spawn (0 <= x0 - $068A < 256): %d, off screen: %d; spawn site exec=%d; first spawn frames: %s", #done, nfull, xmin, xmax, onscreen, offscreen, X.spawnSite,
      table.concat(spawnF, ",", 1, math.min(#spawnF, 14)))
    local keys = {}
    for k in pairs(byChain) do keys[#keys + 1] = k end
    table.sort(keys)
    for _, k in ipairs(keys) do
      local b = byChain[k]
      local lmin, lmax = 999999, 0
      for _, v in ipairs(b.life) do lmin = math.min(lmin, v); lmax = math.max(lmax, v) end
      L("chain %s: n=%d finished=%d life min=%s max=%s steps(first) %s", k, b.n, #b.life, #b.life > 0 and lmin or "-", lmax, b.rle)
    end
    return { lines = m.lines }
  end, { out = o.out })

-- ------------------------------------------------------------------------------------------------ generic room entry
-- n3r.room{ pre = fn(m), scene = nn, variant = v, pick = fn(m) -> slot, dy = 16, settle = 150, arriveSettle = 90, body = fn(m, c), execs = {name = offset}, out = "N3_x.txt" }
--   pre(m)      pokes before the scene warp (disclosed in the run's log)
--   scene       quick warp through sceneNext ($0642) with m1.warp (banner dismissed with A pulses)
--   variant     portal $24 value wanted (default: first portal); pick(m) overrides the slot choice
--   body(m, c)  runs in E6 after the entry; c = { L = log, X = exec counters, S = {gsi rows}, entryF = frame of StartWarp }
function n3r.room(o)
  o = o or {}
  local S = {}
  n3.watchGsi(S)
  local ex = { startWarp = 579376 }
  for k, v in pairs(o.execs or {}) do ex[k] = v end
  local X = n3.execs(ex)
  m1.run(function(m)
    local L = m.log
    if o.pre then o.pre(m) end
    if o.scene then
      local used = m1.warp(o.scene, o.warpSettle or 60)
      L("warp to scene %02X: frames used=%s scene now=%02X", o.scene, tostring(used), kit.r16(0x0644))
    end
    local s
    if o.pick then s = o.pick(m) else
      for _, c in ipairs(n1.live()) do
        local r = n1.rom(n1.ent(c))
        if r >= n3.PORTAL_LO and r <= n3.PORTAL_HI and (o.variant == nil or kit.r16(c + 0x24) == o.variant) then s = c; break end
      end
    end
    if not s then L("no portal (variant %s) in scene %02X", tostring(o.variant), kit.r16(0x0644)); return { lines = m.lines } end
    local px, py = kit.r16(s + 0x14), kit.r16(s + 0x16)
    L("scene=%02X portal slot=%04X entry=%d pos=(%d,%d) $24=%04X $0E=%04X; player form=%d abilities $0AA2=%04X $0AAC=%04X $0B12=%04X", kit.r16(0x0644), s, n1.rom(n1.ent(s)), px, py,
      kit.r16(s + 0x24), kit.r16(s + 0x0E), kit.r16(0x0AD4), kit.r16(0x0AA2), kit.r16(0x0AAC), kit.r16(0x0B12))
    n1.teleport(px, py + (o.dy or 16))
    m.wait(o.settle or 150)
    gaia.hold({ "up" })
    local n = 0
    while X.startWarp == 0 and n < 60 do m.wait(1); n = n + 1 end
    gaia.release()
    L("Up held %d frames before StartWarp (exec=%d)", n, X.startWarp)
    local g = 0
    while g < 1500 and not (kit.r16(0x0644) == 0xE6 and kit.r16(0x0642) == 0 and emu.getState()["ppu.screenBrightness"] == 15) do m.wait(1); g = g + 1 end
    L("arrived in E6 after %d frames: variant $0AAC=%04X src=%04X form=%d $0AB2=%04X banner $B4=%02X", g, kit.r16(0x0AAC), kit.r16(0x0B12), kit.r16(0x0AD4), kit.r16(0x0AB2), kit.r8(0x00B4))
    m.wait(o.arriveSettle or 90)
    local pulse = 0
    n = 0
    while kit.r8(0x00B4) ~= 0 and n < 400 do pulse = pulse + 1; gaia.hold((pulse % 6 < 3) and { "a" } or {}); m.wait(1); n = n + 1 end
    gaia.release(); m.wait(20)
    L("E6 ready: player=(%d,%d) form=%d $0AAC=%04X $06F2=%04X", kit.r16(0x1014), kit.r16(0x1016), kit.r16(0x0AD4), kit.r16(0x0AAC), kit.r16(0x06F2))
    local res = o.body and o.body(m, { L = L, X = X, S = S }) or {}
    res.lines = m.lines
    res.gsi, res.gsiN, res.gsiRows, res.gsiR = S.gsi, S.gsiN, S.gsiRows, S.gsiR
    return res
  end, { out = o.out })
end

-- statue test body for the ability rooms (OQ-I056-1): walks to the statue zone (BranchIfPlayerInAbsTiles ( 3, A, 5, B ): inside when 48 < X <= 80 and 160 < Y - 8 <= 176, in pixels), then logs the player's
-- animation index $28 / $2A, entry word, form, playerFlags $0800 on every change until the Restore routine has run and the ability grant has started.
function n3r.statue(o)
  o = o or {}
  return function(m, c)
    local L = c.L
    local X = c.X
    local pa = kit.r16(0x09AA)
    local imgs, last, pend = {}, nil, {}
    local function row(t, why)
      local key = string.format("$28=%04X ent=%d form=%d pf0800=%d abil=%04X bgm=%04X", kit.r16(pa + 0x28), n1.rom(n1.ent(pa)), kit.r16(0x0AD4), (kit.r16(0x09AE) & 0x0800) ~= 0 and 1 or 0, kit.r16(0x0AA2), kit.r16(0x06F2))
      if key ~= last then
        last = key
        c.nrow = (c.nrow or 0) + 1
        if c.nrow <= (o.maxRows or 60) then L("t=%d %s %s mask=%04X $0AAC=%04X", t, why or "", key, kit.r16(0x065A), kit.r16(0x0AAC)) end
        if #imgs + #pend < (o.maxGrabs or 12) and X.dispatch > 0 and t > 0 then pend[#pend + 1] = t + 3 end -- the staged frame is on screen 2-3 frames later
      end
    end
    local function grabs(k)
      if pend[1] and pend[1] <= k then
        local cr = o.crop or {}
        imgs[#imgs + 1] = n2.grab(o.step or 2, cr[1], cr[2], cr[3], cr[4])
        table.remove(pend, 1)
        L("grab #%d at t=%d", #imgs, k)
      end
    end
    -- walk: left until x <= 76, up until y <= 180
    local t = 0
    gaia.hold({ "left" })
    local g = 0
    while kit.r16(0x1014) > 76 and g < 120 do m.wait(1); g = g + 1 end
    gaia.release(); m.wait(1)
    gaia.hold({ "up" })
    g = 0
    while kit.r16(0x1016) > 180 and g < 120 and X.dispatch == 0 do m.wait(1); g = g + 1 end
    gaia.release()
    L("walk done: player=(%d,%d) dispatch exec=%d", kit.r16(0x1014), kit.r16(0x1016), X.dispatch)
    g = 0
    while X.dispatch == 0 and g < 60 do m.wait(1); g = g + 1 end
    L("dispatch exec=%d after %d more frames", X.dispatch, g)
    row(0, "pre")
    local pulse = 0
    for k = 1, o.frames or 700 do
      m.wait(1); t = k
      row(k); grabs(k)
      if k > (o.pulseFrom or 400) then pulse = pulse + 1; gaia.hold((pulse % 8 < 3) and { "a" } or {}) end
    end
    gaia.release()
    local bits = {}
    for k, v in pairs(X) do bits[#bits + 1] = k .. "=" .. v end
    table.sort(bits)
    L("exec counts: %s", table.concat(bits, " "))
    L("end: form=%d abil=%04X $0AAC=%04X player entry=%d mask=%04X pf=%04X $06F2=%04X", kit.r16(0x0AD4), kit.r16(0x0AA2), kit.r16(0x0AAC), n1.rom(n1.ent(pa)), kit.r16(0x065A), kit.r16(0x09AE), kit.r16(0x06F2))
    local out = {}
    if #imgs > 0 then out.sheetBytes = n3.sheet(imgs, math.min(o.cols or 4, #imgs), o.name or "statue", o.scale) end
    return out
  end
end

-- ------------------------------------------------------------------------------------------------ Gaia talk body (OQ-I056-4)
-- walks the player up into Gaia's zone (BranchIfPlayerInAbsTiles ( 7, 8, 9, 9 ): 112 < X <= 144 and 128 < Y - 8 <= 144), pulses A (physical A = dialogue advance) and logs every exec counter change
-- with $0AAC, the inventory string and the dialogue text candidates, until GaiaSavePrompt has run (then `after` more frames).

      end

-- ------------------------------------------------------------------------------------------------ Dark Gaia projectile sprites (OQ-I067-3)
-- Scene $E8: any melee attack spawns a Projectile* actor (player_character.asm, SpawnListAppend flags $0602). Projectile entries (ROM offsets of the first instruction): South 184168, North 184194, West 184220, East 184246.
-- For each attack direction: hold the direction 10 frames, press physical B (interact / attack) for 2 frames, then sample 45 frames. On every change of the projectile's ($28, $2A) log the slot position and the OAM rows that are
-- NOT in the baseline (taken before the attack): x, y, tile (OAM byte 2), attribute (byte 3: bit 0 = name table, bits 1-3 = palette), size bit; grab a crop around the new sprites.
n3r.PROJ_LO, n3r.PROJ_HI = 184168, 184290
function n3r.projectile(o)
  o = o or {}
  local SPR = emu.memType.snesSpriteRam
  local function oam()
    local t = {}
    for i = 0, 127 do
      local b = i * 4
      local hi = emu.read(512 + (i >> 2), SPR)
      local sh = (i & 3) * 2
      local x = emu.read(b, SPR) | (((hi >> sh) & 1) << 8)
      t[i] = { x = x, y = emu.read(b + 1, SPR), tile = emu.read(b + 2, SPR), attr = emu.read(b + 3, SPR), big = (hi >> (sh + 1)) & 1 }
    end
    return t
  end
  local function key(e) return string.format("%02X/%02X", e.tile, e.attr) end
  m1.run(function(m)
    local L = m.log
    local u = m1.warp(0xE8, o.settle or 300)
    if o.tx then n1.teleport(o.tx, o.ty); m.wait(60) end
    L("warp to E8: frames used=%s scene=%02X form=%d player=(%d,%d) hp=%d", tostring(u), kit.r16(0x0644), kit.r16(0x0AD4), kit.r16(0x1014), kit.r16(0x1016), kit.r16(0x0ACE))
    local st = emu.getState()
    L("OAM: base words=%s name gap words=%s size mode=%s (state keys ppu.oam*)", tostring(st["ppu.oamBaseAddress"]), tostring(st["ppu.oamAddressOffset"]), tostring(st["ppu.oamMode"]))
    local imgs, n = {}, 0
    local seenKey = {}
    for _, d in ipairs(o.dirs or { "down", "up", "left", "right" }) do
      gaia.hold({ d }); m.wait(3); gaia.release(); m.wait(8)
      local base, bk = oam(), {}
      for i = 0, 127 do if base[i].y < 224 then bk[key(base[i])] = true end end
      m1.press({ "b" }, 3)
      local last = nil
      for t = 1, 45 do
        m.wait(1)
        for _, s in ipairs(n1.live()) do
          local r = n1.rom(n1.ent(s))
          if r >= n3r.PROJ_LO and r <= n3r.PROJ_HI then
            local k = string.format("%04X:%04X/%04X", s, kit.r16(s + 0x28), kit.r16(s + 0x2A))
            if k ~= last then
              last = k
              local nm = (r < 184194 and "South") or (r < 184220 and "North") or (r < 184246 and "West") or "East"
              local rows, minx, miny = {}, 999, 999
              local cur = oam()
              for i = 0, 127 do
                local e = cur[i]
                if e.y < 224 and not bk[key(e)] then
                  rows[#rows + 1] = string.format("%d,%d t%02X a%02X%s", e.x, e.y, e.tile, e.attr, e.big == 1 and "L" or "s")
                  minx = math.min(minx, e.x); miny = math.min(miny, e.y)
                  seenKey[key(e)] = true
                end
              end
              L("dir=%s t=%d Projectile%s slot=%04X pos=(%d,%d) $28=%04X $2A=%04X $10=%04X new OAM(%d): %s", d, t, nm, s, kit.r16(s + 0x14), kit.r16(s + 0x16), kit.r16(s + 0x28), kit.r16(s + 0x2A), kit.r16(s + 0x10), #rows, table.concat(rows, " | "):sub(1, 400))
              if #rows > 0 and n < (o.maxGrabs or 16) then
                local gx, gy = math.max(0, minx - 6), math.max(0, miny - 6)
                imgs[#imgs + 1] = n2.grab(1, gx, gy, 64, 64)
                n = n + 1
              end
            end
          end
        end
      end
      m.wait(60)
    end
    local ks = {}
    for k in pairs(seenKey) do ks[#ks + 1] = k end
    table.sort(ks)
    L("tile/attr pairs new in the OAM during the projectile frames: %s", table.concat(ks, " "))
    local res = { lines = m.lines }
    if #imgs > 0 then res.sheetBytes = n3.sheet(imgs, 4, o.name or "projectile", 3) end
    return res
  end, { out = o.out })
end

-- ------------------------------------------------------------------------------------------------ VRAM DMA census for the sprite name table 0 ($4000-$4FFF words) and 1 ($5000-$5FFF), E8 arrival
-- Logs every general DMA ($420B) whose B-bus destination is $2118/$2119 with the VRAM word address current at the start, the source, the byte count and the PC after the store. Only transfers that touch $4000-$5FFF are logged.
function n3r.vramDma(o)
  o = o or {}
  local rows = {}
  local function key(st, c)
    local p = "dmaController.channel[" .. c .. "]."
    return st[p .. "srcBank"] or 0, st[p .. "srcAddress"] or 0, st[p .. "destAddress"] or 0, st[p .. "transferSize"] or 0
  end
  local vkey
  local function onMdma(addr, v)
    local st = emu.getState()
    if not vkey then
      for k in pairs(st) do if k:lower():find("vramaddress") and not k:lower():find("remap") and not k:lower():find("increment") then vkey = k; break end end
    end
    for c = 0, 7 do
      if (v >> c) & 1 == 1 then
        local sb, sa, da, sz = key(st, c)
        local va = vkey and st[vkey] or -1
        if da == 0x18 and va >= 0x4000 and va < 0x6000 and #rows < 60 then
          rows[#rows + 1] = string.format("f=%d vram=%04X src=%02X:%04X bytes=%d pc=%02X:%04X", st.frameCount, va, sb, sa, sz, st["cpu.k"], st["cpu.pc"])
        end
      end
    end
  end
  for bank = 0, 0xBF do
    if bank <= 0x3F or bank >= 0x80 then
      emu.addMemoryCallback(onMdma, emu.callbackType.write, (bank << 16) | 0x420B, (bank << 16) | 0x420B, emu.cpuType.snes, emu.memType.snesMemory)
    end
  end
  m1.run(function(m)
    local L = m.log
    m1.warp(0xE8, o.settle or 300)
    L("vram address key=%s", tostring(vkey))
    for i, r in ipairs(rows) do L("%d %s", i, r) end
    return { lines = m.lines }
  end, { out = o.out })
end

-- Compare VRAM word $4400.. (byte $8800, 1536 bytes) with ROM 1886848.. (misc_fx_1CCA80) in scene E8 after arrival; also count the tiles $40-$6F that differ from the 1536 ROM bytes and list tile-by-tile non-blank state.
function n3r.projVram(o)
  o = o or {}
  m1.run(function(m)
    local L = m.log
    m1.warp(0xE8, o.settle or 300)
    local V, R = emu.memType.snesVideoRam, emu.memType.snesPrgRom
    local diff, firstDiff = 0, nil
    for i = 0, 1535 do
      if emu.read(0x8800 + i, V) ~= emu.read(1886848 + i, R) then
        diff = diff + 1; firstDiff = firstDiff or i
      end
    end
    L("scene=%02X playerFlags=%04X: VRAM bytes $8800..$8DFF vs ROM 1886848..1888383: %d differ (first %s)", kit.r16(0x0644), kit.r16(0x09AE), diff, tostring(firstDiff))
    local blank = {}
    for t = 0x40, 0x6F do
      local nz = 0
      for i = 0, 31 do if emu.read(0x8000 + t * 32 + i, V) ~= 0 then nz = nz + 1 end end
      blank[#blank + 1] = string.format("%02X:%d", t, nz)
    end
    L("non-zero bytes per tile (of 32): %s", table.concat(blank, " "))
    local pal = {}
    for c = 0xA0, 0xAF do pal[#pal + 1] = string.format("%04X", emu.read(c * 2, emu.memType.snesCgRam) | (emu.read(c * 2 + 1, emu.memType.snesCgRam) << 8)) end
    L("CGRAM OBJ palette 2 (colours $A0-$AF): %s", table.concat(pal, " "))
    return { lines = m.lines }
  end, { out = o.out })
end
```
