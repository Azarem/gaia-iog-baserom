## kit/lib/fr10_kit.lua

```lua
-- description: FR10 early-game capture helpers: segment logger (flag rows, scene changes, input-lock changes), state snapshot, and an input bot (step program, collision-grid walker, run-length recording of the buttons it pressed).
-- requires: gaia_kit
--
-- Load with run_script / load_script  includes=["fr10_kit"]   (the server resolves "requires": gaia_kit, mesen_kit, iog_kit).
-- All addresses are WRAM offsets. Nothing here writes game memory.
--
--   fr10.snap()                          {f, sc, next, mask, banner, x, y, ax, ay, hp, flags10}   (x, y = $09A2/$09A4, ax, ay = actor position)
--   fr10.start{cap=200, sceneCap=120, maskCap=60, minMaskFrames=1}
--        -> handle h with h:summary(): starts gaia.flagwatch over $0A00-$0AB3 with its own log (rows past `cap` are counted),
--           a SCENE row at every change of sceneCurrent and a MASK row at every change of joypadMaskStd between zero and nonzero.
--   fr10.bot{ steps = {...}, maxFrames = N, onDone = fn(result) }   input program (below); the run ends with kit.finish(result)
--        steps (each ends when its condition holds or `max` frames pass; `max` defaults to 600):
--          {hold = {"a"}, n = 10}                     hold the buttons for n frames
--          {wait = 30}                                no buttons for n frames
--          {waitUntil = fn, max = 600}                no buttons until fn() is true (false after `max` frames)
--          {dir = "right", untilFn = fn, max = 300}   hold a direction until untilFn() is true
--          {mash = "a", down = 2, gap = 6, untilFn = fn, max = 600, onlyDialog = false}   press `mash` for `down` frames, release for `gap`, until untilFn()
--                                                    (onlyDialog = true presses only while a dialog or the scene banner is up)
--          {walk = {px, py}, tol = 1, max = 900, pass = {[0]=true}, ledge = true, soft = false, dismiss = false}   walk the player point ($09A2/$09A4, the top-left
--                                                    corner of a 16 x 16 hitbox) to (px, py): BFS over the collision cells (low nibble in
--                                                    `pass`, default {0}; high nibble must be 0), replanned whenever the player enters a new
--                                                    cell (tol = 0 for an exact trigger position; the final tolerance is 1 px by default, the
--                                                    intermediate one 1 px as well). Cells with low nibble 8 are ledges (stepping on one locks
--                                                    the joypad and moves the player to another level): they are never crossed by the plan, but
--                                                    when the target is unreachable the walker steps onto the nearest reachable ledge (ledge =
--                                                    false disables). Slope cells (5, 6, 9) are walked with both axes pressed (vertical-only
--                                                    presses do nothing on them). A joypad lock pauses the walk; with dismiss = true a dialog that
--                                                    opens on the way is dismissed with A (without it the walk keeps pressing directions). A scene change ends the step as ok; 40 frames without movement or `max`
--                                                    frames fail it, and a failed step ends the whole program (reason step_failed) unless `soft`.
--          {call = fn}                                run fn(snap) once (log or record something), no input for that frame
--        result: {rle = "frames:button+button ...", totalFrames, steps = {rows}, reason, final}
fr10 = fr10 or {}

local function r16(a) return kit.r16(a) end

function fr10.snap()
  return {
    f = emu.getState().frameCount, sc = r16(0x0644), next = r16(0x0642), mask = r16(0x065A), banner = kit.r8(0x00B4),
    x = r16(0x09A2), y = r16(0x09A4), ax = r16(0x1014), ay = r16(0x1016), hp = r16(0x0ACE), flags10 = r16(0x1010),
  }
end

-- segment logger ------------------------------------------------------------------------------------
function fr10.start(opts)
  opts = opts or {}
  local cap, sceneCap, maskCap = opts.cap or 200, opts.sceneCap or 120, opts.maskCap or 60
  local nflag, nscene, nmask = 0, 0, 0
  local function flagLog(fmt, ...)
    if nflag < cap then nflag = nflag + 1; emu.log(string.format(fmt, ...)) end
  end
  local h = { fw = gaia.flagwatch{ cap = cap, log = flagLog, actor = true }, scenes = 0, masks = 0 }
  local lastScene, lastMaskNz = r16(0x0644), r16(0x065A) ~= 0
  kit.onFrame(function()
    local sc = r16(0x0644)
    if sc ~= lastScene then
      h.scenes = h.scenes + 1
      if nscene < sceneCap then
        nscene = nscene + 1
        local s = fr10.snap()
        emu.log(string.format("SCENE f=%d $%02X>$%02X next=%04X mask=%04X banner=%02X pos=%d,%d", s.f, lastScene & 0xFF, sc & 0xFF, s.next, s.mask, s.banner, s.ax, s.ay))
      end
      lastScene = sc
    end
    local nz = r16(0x065A) ~= 0
    if nz ~= lastMaskNz then
      h.masks = h.masks + 1
      if nmask < maskCap then
        nmask = nmask + 1
        local s = fr10.snap()
        emu.log(string.format("MASK f=%d sc=%02X %s mask=%04X pos=%d,%d", s.f, sc & 0xFF, nz and "LOCK" or "FREE", s.mask, s.ax, s.ay))
      end
      lastMaskNz = nz
    end
  end)
  function h:summary()
    local s = self.fw:summary()
    s.sceneChanges, s.maskChanges = self.scenes, self.masks
    return s
  end
  return h
end

-- bot ---------------------------------------------------------------------------------------------------
local function buttonsKey(t)
  local names = {}
  for k, v in pairs(t) do if v then names[#names + 1] = k end end
  table.sort(names)
  return table.concat(names, "+"), names
end

local function isFree(v, pass) return pass[v & 0x0F] and (v >> 4) == 0 end
local SLOPE = { [5] = true, [6] = true, [9] = true }
-- on a slope only horizontal buttons move the player: low nibble 9 = right goes down-right (left up-left), 6 = left goes down-left (right up-right); 5 is not probed yet
local SLOPE_VEC = { [9] = { 1, 1 }, [6] = { -1, 1 } }
local function isSlope(cx, cy) return SLOPE[gaia.coll.get(cx, cy) & 0x0F] == true end

-- BFS over the collision cells of the scene from cell (sx, sy) to (tx, ty); returns the list of cells (excluding the start) or nil.
-- Returns (path, kind): kind "target" = the path ends on the target cell; "ledge" = the target is unreachable and the path ends on the nearest ledge cell.
local function bfs(sx, sy, tx, ty, pass, ledge)
  local stride, rows = kit.r8(0x0693), kit.r8(0x0697)
  local w, h = stride * 16, rows * 16
  if sx < 0 or sy < 0 or sx >= w or sy >= h or tx < 0 or ty < 0 or tx >= w or ty >= h then return nil end
  local seen, prev, queue, qi = {}, {}, { sy * w + sx }, 1
  seen[sy * w + sx] = true
  local target = ty * w + tx
  local ledges = {}
  local function build(cur)
    local path, c = {}, cur
    while c ~= sy * w + sx do table.insert(path, 1, c); c = prev[c] end
    local out = {}
    for _, id in ipairs(path) do out[#out + 1] = { id % w, id // w } end
    return out
  end
  while qi <= #queue do
    local cur = queue[qi]; qi = qi + 1
    if cur == target then return build(cur), "target" end
    local cx, cy = cur % w, cur // w
    for _, d in ipairs { { 1, 0 }, { -1, 0 }, { 0, 1 }, { 0, -1 } } do
      local nx, ny = cx + d[1], cy + d[2]
      if nx >= 0 and ny >= 0 and nx < w and ny < h then
        local id = ny * w + nx
        if not seen[id] then
          local v = gaia.coll.get(nx, ny)
          if id == target or isFree(v, pass) then
            seen[id] = true; prev[id] = cur; queue[#queue + 1] = id
          elseif ledge and (v & 0x0F) == 8 and (v >> 4) == 0 then
            seen[id] = true; prev[id] = cur
            ledges[#ledges + 1] = id
          end
        end
      end
    end
  end
  -- unreachable target: use the reachable ledge whose landing (assumed 8 cells below the ledge, as at the church and the town streets) is nearest to the
  -- target, but only when that is nearer than the start cell (no ping-pong between levels)
  local best, bestScore = nil, math.abs(sx - tx) + math.abs(sy - ty)
  for _, id in ipairs(ledges) do
    -- stepping onto a ledge from above drops the player 8 cells down, from below it lifts the player 8 cells up (horizontal approaches are not used)
    local ly, fromY = id // w, prev[id] // w
    local landing = (fromY < ly) and (ly + 8) or ((fromY > ly) and (ly - 8) or nil)
    if landing then
      local score = math.abs(id % w - tx) + math.abs(landing - ty)
      if score < bestScore then best, bestScore = id, score end
    end
  end
  if best then return build(best), "ledge" end
  return nil
end

function fr10.bot(o)
  local steps = o.steps
  local rle, total = {}, 0
  local stepLog = {}
  local si, sf = 0, 0            -- current step index, frames in the step
  local cur, state = nil, {}
  local lastKey, lastNames
  local finished = false

  local function finish(reason)
    if finished then return end
    finished = true
    -- compact result (the tool reply is read by a person): rle as "frames:button+button" words ("-" = no button), steps as text rows
    local words = {}
    for _, e in ipairs(rle) do words[#words + 1] = e.frames .. ":" .. (#e.buttons > 0 and table.concat(e.buttons, "+") or "-") end
    local rows = {}
    for _, s in ipairs(stepLog) do
      local sn = s.snap or {}
      rows[#rows + 1] = string.format("step %s ok=%s %s frames=%s f=%s sc=%s pos=%s,%s mask=%04X", tostring(s.i), tostring(s.ok), s.reason or "", tostring(s.frames), tostring(sn.f), sn.sc and string.format("%02X", sn.sc) or "?", tostring(sn.ax), tostring(sn.ay), sn.mask or 0)
    end
    local fin = fr10.snap()
    local res = { rle = table.concat(words, " "), totalFrames = total, steps = rows, reason = reason,
                  final = string.format("f=%d sc=%02X next=%04X mask=%04X banner=%02X pos=%d,%d player09A2=%d,%d hp=%d", fin.f, fin.sc, fin.next, fin.mask, fin.banner, fin.ax, fin.ay, fin.x, fin.y, fin.hp) }
    if o.onDone then o.onDone(res) end
    kit.finish(res)
  end

  local function beginStep()
    si = si + 1
    sf = 0
    state = {}
    cur = steps[si]
    if not cur then finish("steps_done") end
  end

  local function endStep(ok)
    stepLog[#stepLog + 1] = { i = si, ok = ok, frames = sf, snap = fr10.snap() }
    -- a failed step ends the program (the later steps assume its result) unless it is marked soft = true
    if not ok and not (cur and cur.soft) then finish("step_failed"); cur = nil; return end
    beginStep()
  end

  beginStep()

  emu.addEventCallback(function()
    if finished then return end
    if o.maxFrames and total >= o.maxFrames then finish("maxFrames"); return end
    local t = {}
    local guard = 0
    -- a step that finishes without consuming a frame lets the next one decide in the same poll
    while cur and guard < 8 do
      guard = guard + 1
      local max = cur.max or 600
      if cur.hold then
        if sf >= cur.n then endStep(true)
        else for _, b in ipairs(cur.hold) do t[b] = true end; break end
      elseif cur.waitUntil then
        if cur.waitUntil() then endStep(true)
        elseif sf >= max then endStep(false)
        else break end
      elseif cur.wait then
        if sf >= cur.wait then endStep(true) else break end
      elseif cur.call then
        cur.call(fr10.snap()); endStep(true)
      elseif cur.dir then
        if cur.untilFn and cur.untilFn() then endStep(true)
        elseif sf >= max then endStep(false)
        else t[cur.dir] = true; break end
      elseif cur.mash then
        if cur.untilFn and cur.untilFn() then endStep(true)
        elseif sf >= max then endStep(false)
        else
          local period = (cur.down or 2) + (cur.gap or 6)
          -- onlyDialog: press only while a dialog (displayModeFlags bit 13) or the scene banner is up, so the press that closes a dialog is not repeated into a new interaction
          local active = (not cur.onlyDialog) or (r16(0x09EC) & 0x2000) ~= 0 or kit.r8(0x00B4) ~= 0
          if active and (sf % period) < (cur.down or 2) then t[cur.mash] = true end
          break
        end
      elseif cur.walk then
        local pass = cur.pass or { [0] = true }
        local tol = cur.tol or 1
        local px, py = cur.walk[1], cur.walk[2]
        local x, y = r16(0x09A2), r16(0x09A4)
        -- replan whenever the player enters a new cell (ledges move the player to another level); a locked joypad (ledge drop, dialog) pauses the walk
        local cellKey = (x >> 4) * 4096 + (y >> 4)
        state.scene = state.scene or r16(0x0644)
        if r16(0x0644) ~= state.scene or r16(0x0642) ~= 0 then endStep(true); break end   -- a warp fired on the way: the walk is over
        if cur.dismiss and (r16(0x09EC) & 0x2000) ~= 0 then
          -- dismiss = true: a dialog opened on the way (an NPC or a position trigger) is dismissed with A, then the walk carries on
          if sf >= max then endStep(false); break end
          if (sf % 22) < 2 then t.a = true end
          state.cellKey = nil; state.lastPos = nil
          break
        end
        if kit.r16(0x065A) ~= 0 then
          if sf >= max then endStep(false); break end
          state.cellKey = nil; state.lastPos = nil
          break
        end
        if state.cellKey ~= cellKey then
          state.cellKey = cellKey
          state.path, state.kind = bfs(x >> 4, y >> 4, px >> 4, py >> 4, pass, cur.ledge ~= false)
          if cur.trace then
            local p = state.path
            emu.log(string.format("walk plan from p=%d,%d cell=%d,%d kind=%s len=%s next=%s last=%s", x, y, x >> 4, y >> 4, tostring(state.kind), p and #p or "nil", p and p[1] and (p[1][1] .. "," .. p[1][2]) or "-", p and p[#p] and (p[#p][1] .. "," .. p[#p][2]) or "-"))
          end
        end
        if not state.path then
          -- right after a scene change the player point can still be stale: retry the plan for up to 90 frames before giving up
          state.noPath = (state.noPath or 0) + 1
          if state.noPath < 90 then
            state.cellKey = nil
            break
          end
          stepLog[#stepLog + 1] = { i = si, ok = false, reason = "no_path", snap = fr10.snap() }
          endStep(false)
        elseif sf >= max then endStep(false)
        else
          local path = state.path
          local wp = path[1]
          local gx, gy
          if wp then
            -- the player point ($09A2/$09A4) is the top-left corner of a 16 x 16 hitbox: a cell is occupied exactly when the point is a multiple of 16
            if #path == 1 and state.kind == "target" then gx, gy = px, py else gx, gy = wp[1] * 16, wp[2] * 16 end
          else
            gx, gy = px, py
          end
          local dx, dy = gx - x, gy - y
          local lastWp = (not wp) or (#path == 1 and state.kind == "target")
          local ttol = lastWp and tol or 1
          if math.abs(dx) <= ttol and math.abs(dy) <= ttol then
            if lastWp then endStep(true)
            else
              -- waypoint reached (the player can be within tolerance without having changed cell): advance along the plan, replan once it is used up
              table.remove(state.path, 1)
              if #state.path == 0 then state.cellKey = nil end
            end
          else
            if kit.r16(0x065A) == 0 then
              -- the hitbox is 16 x 16: to move along a row the player must first be aligned to that row (and to a column when moving along a column)
              local horizontal = wp and wp[2] == (y >> 4)
              local needX, needY = math.abs(dx) > ttol, math.abs(dy) > ttol
              -- on slope cells (low nibble 5, 6, 9) the engine moves the player along the slope: no alignment, press the direction of the next cell and replan on every cell change
              local slopeCell = wp and (isSlope(x >> 4, y >> 4) and gaia.coll.get(x >> 4, y >> 4) or isSlope((x + 15) >> 4, y >> 4) and gaia.coll.get((x + 15) >> 4, y >> 4)
                or isSlope(x >> 4, (y + 15) >> 4) and gaia.coll.get(x >> 4, (y + 15) >> 4) or isSlope((x + 15) >> 4, (y + 15) >> 4) and gaia.coll.get((x + 15) >> 4, (y + 15) >> 4))
              if wp and (slopeCell or isSlope(wp[1], wp[2])) then
                -- slope mode: no alignment; press both axes toward the first path cell at least 24 px away (a slope band is diagonal) and replan on every cell change
                local L = path[#path]
                for i = 1, #path do
                  local c = path[i]
                  if math.max(math.abs(c[1] * 16 - x), math.abs(c[2] * 16 - y)) >= 24 then L = c; break end
                end
                local ldx, ldy = L[1] * 16 - x, L[2] * 16 - y
                local pressedH = false
                if ldx >= 2 then t.right = true; pressedH = true elseif ldx <= -2 then t.left = true; pressedH = true end
                if ldy >= 2 then t.down = true elseif ldy <= -2 then t.up = true end
                local vec = slopeCell and SLOPE_VEC[slopeCell & 0x0F]
                if not pressedH and vec and (ldy >= 2 or ldy <= -2) then
                  -- a vertical step on a slope: only a horizontal button moves the player, along the slope vector (9: right = down-right, 6: left = down-left)
                  local dirX = (ldy > 0) and vec[1] or -vec[1]
                  t[dirX > 0 and "right" or "left"] = true
                end
              else
                if horizontal and needY then t[dy > 0 and "down" or "up"] = true
                elseif needX then t[dx > 0 and "right" or "left"] = true
                elseif needY then t[dy > 0 and "down" or "up"] = true end
                -- an odd offset of 1-2 px on the column cannot be closed by a single-axis press (2 px steps): carry the row movement along with it
                if needX and needY and math.abs(dx) <= 2 and not horizontal then t[dy > 0 and "down" or "up"] = true end
                -- fine alignment: a diagonal press moves the other axis by 1 or 2 px per frame (single-axis steps are 2 px, so an odd position never aligns);
                -- a small perpendicular offset is corrected together with the main direction (the main axis may be blocked by that 1-2 px overlap)
                if horizontal then
                  if dy ~= 0 and math.abs(dy) <= 2 and not t.up and not t.down then t[dy > 0 and "down" or "up"] = true end
                elseif wp then
                  if dx ~= 0 and math.abs(dx) <= 2 and not t.left and not t.right then
                    -- only when the cells the hitbox would enter are walkable: a diagonal press into a wall or slope blocks the main axis too
                    local nx = (dx > 0) and ((x + 17) >> 4) or ((x - 2) >> 4)
                    local ok = true
                    for _, yy in ipairs { y, y + 15 } do
                      local v = gaia.coll.get(nx, yy >> 4)
                      if not (pass[v & 15] and (v >> 4) == 0) or isSlope(nx, yy >> 4) then ok = false end
                    end
                    if ok then t[dx > 0 and "right" or "left"] = true end
                  end
                end
              end
            end
            -- stall guard: no movement for 40 frames while the joypad is not locked
            state.lastPos = state.lastPos or { x, y, sf }
            if state.lastPos[1] ~= x or state.lastPos[2] ~= y or kit.r16(0x065A) ~= 0 then state.lastPos = { x, y, sf } end
            if sf - state.lastPos[3] > 40 then stepLog[#stepLog + 1] = { i = si, ok = false, reason = "stalled", snap = fr10.snap() }; endStep(false) end
            break
          end
        end
      else
        endStep(false)
      end
    end
    emu.setInput(t, 0)
    if not cur then
      -- steps ended during this poll: record nothing further
      return
    end
    local key, names = buttonsKey(t)
    if key == lastKey and #rle > 0 then rle[#rle].frames = rle[#rle].frames + 1
    else rle[#rle + 1] = { frames = 1, buttons = names }; lastKey = key end
    total = total + 1
    sf = sf + 1
  end, emu.eventType.inputPolled)
end
```

## kit/lib/fr10_steps.lua

```lua
-- description: FR10 common step conditions for fr10.bot programs (free-control test, busy test, scene tests, position test, flag test, any-of combiner).
-- requires: fr10_kit
--
--   fr10s.free(n)            untilFn: the player is free (joypadMaskStd == 0, banner byte == 0, no dialogue mode, no script lock, scene loaded: flag $FF set, no scene change queued) for n polls in a row (default 20)
--   fr10s.busy()             untilFn: the player is not free right now (joypad mask, banner, dialogue mode or script lock)
--   fr10s.scene(id)          untilFn: sceneCurrent == id and sceneNext == 0
--   fr10s.leaves(id)         untilFn: sceneCurrent ~= id or a scene change is queued
--   fr10s.at(px, py, tol)    untilFn: the player point ($09A2/$09A4) is within tol (default 0) of (px, py)
--   fr10s.flag(n)            untilFn: global flag n is set
--   fr10s.any(f1, f2, ...)   untilFn: true when any of the functions is true (all are evaluated every poll, so stateful ones such as free(n) keep counting)
fr10s = fr10s or {}

local function isFreeNow()
  -- a running dialog clears joypadMaskStd while it is on screen: displayModeFlags bit 13 (dialogue mode) and playerFlags bit 11 (script lock) are part of the test
  -- flag $FF ($0A1F bit 7) is cleared when a scene starts to load and set again by the scene's first actor once the scene runs: part of the test so a half-loaded scene is not "free"
  return kit.r16(0x065A) == 0 and kit.r8(0x00B4) == 0 and (kit.r16(0x09EC) & 0x2000) == 0 and (kit.r16(0x09AE) & 0x0800) == 0
    and (kit.r8(0x0A1F) & 0x80) ~= 0 and kit.r16(0x0642) == 0
end

function fr10s.free(n)
  n = n or 20
  local run = 0
  return function()
    if isFreeNow() then run = run + 1 else run = 0 end
    return run >= n
  end
end

function fr10s.busy()
  return function() return not isFreeNow() end
end

function fr10s.scene(id)
  return function() return kit.r16(0x0644) == id and kit.r16(0x0642) == 0 end
end

function fr10s.leaves(id)
  return function() return kit.r16(0x0644) ~= id or kit.r16(0x0642) ~= 0 end
end

function fr10s.at(px, py, tol)
  tol = tol or 0
  return function() return math.abs(kit.r16(0x09A2) - px) <= tol and math.abs(kit.r16(0x09A4) - py) <= tol end
end

function fr10s.atX(px, tol)
  tol = tol or 0
  return function() return math.abs(kit.r16(0x09A2) - px) <= tol end
end

function fr10s.flag(n)
  return function() return gaia.flags.get(n) == 1 end
end

-- step-list helpers ---------------------------------------------------------------------------------

-- concatenate step lists
function fr10s.cat(...)
  local out = {}
  for _, l in ipairs { ... } do for _, s in ipairs(l) do out[#out + 1] = s end end
  return out
end

-- walk to the doorstep cell (cx, cy) below a town door, hold Up (the door opens after 8 frames), dismiss narrations with A, and continue up until the scene is left.
-- `from` = the scene the door is in, `to` = the destination scene. side = -1 / 1: the doorstep cell can be solid (the door's solid mark can cover it), but the door also opens for a
-- player one cell to the west (-1) or east (1) of the doorstep: wait there with Up held, then step onto the doorstep column.
function fr10s.enterDoor(cx, cy, from, to, pass, side)
  local approach
  if side then
    approach = {
      { walk = { (cx + side) * 16, cy * 16 }, pass = pass, max = 2500, tol = 1 },
      { hold = { "up" }, n = 30 },
      { dir = side < 0 and "right" or "left", untilFn = fr10s.atX(cx * 16, 1), max = 400 },
    }
  else
    approach = { { walk = { cx * 16, cy * 16 }, pass = pass, max = 2500 } }
  end
  return fr10s.cat(approach, {
    { dir = "up", untilFn = fr10s.any(fr10s.busy(), fr10s.leaves(from)), max = 240 },
    { mash = "a", down = 2, gap = 20, untilFn = fr10s.any(fr10s.free(30), fr10s.leaves(from)), max = 900, onlyDialog = true },
```

```lua
local CAP, nlog = 200, 0

-- collision layer: 256-byte pages of 16x16 tiles; page = (ty>>4) * stride + (tx>>4); see tile_collision_physics.asm
gaia.coll = { base = 0x1C000 }

function gaia.coll.index(tx, ty)
  local stride = kit.r8(gaia.addr.collStride)
  return (((ty >> 4) * stride + (tx >> 4)) << 8) | ((ty & 15) << 4) | (tx & 15)
end

function gaia.coll.get(tx, ty) return kit.r8(gaia.coll.base + gaia.coll.index(tx, ty)) end
function gaia.coll.set(tx, ty, v) kit.w8(gaia.coll.base + gaia.coll.index(tx, ty), v) end

-- CheckActorOnSpecialTile probes (actorX - 8, actorY - 16); the probe is skipped when either value is 16-aligned.
function gaia.coll.probeTile(x, y)
  local px, py = x - 8, y - 16
  return px >> 4, py >> 4, (px & 15) ~= 0 and (py & 15) ~= 0
end

-- B1 census of the whole collision layer of the loaded scene ------------------------------------
-- Walks every cell of the page grid (stride = $0693 pages across, $0697 pages down; one page = 16 x 16 cells) with
-- gaia.coll.get. Returns one row (a table for kit.finish): bounds words, strides, per-nibble histograms (index = value + 1),
-- the first `maxFirst` cells (tx, ty in row-major order) of every type in `firsts` (low nibble), map-data-size words.
-- A page grid above 64 pages would run past $7F:FFFF (the layer starts at $7F:C000): it is clamped and flagged.
function gaia.coll.census(firsts, maxFirst)
  firsts, maxFirst = firsts or { 6, 9 }, maxFirst or 10
  local A = gaia.addr
  local stride, rows = kit.r8(A.collStride), kit.r8(0x0697)
  local clamped = false
  if stride * rows > 64 then rows = math.max(1, 64 // math.max(stride, 1)); clamped = true end
  local w, h = stride * 16, rows * 16
  local low, hi = {}, {}
  for i = 1, 16 do low[i], hi[i] = 0, 0 end
  local want, first = {}, {}
  for _, t in ipairs(firsts) do want[t] = true; first[string.format("%02X", t)] = {} end
  for ty = 0, h - 1 do
    for tx = 0, w - 1 do
      local v = gaia.coll.get(tx, ty)
      local lo, hn = v & 15, v >> 4
      low[lo + 1] = low[lo + 1] + 1
      hi[hn + 1] = hi[hn + 1] + 1
      if want[lo] then
        local f = first[string.format("%02X", lo)]
        if #f < maxFirst then f[#f + 1] = { tx, ty, v } end
      end
    end
  end
  return {
    scene = kit.r16(A.sceneCur), boundsX = kit.r16(A.boundsX), boundsY = kit.r16(A.boundsY),
    stride = stride, rows = rows, cellsW = w, cellsH = h, clamped = clamped,
    stride1 = kit.r8(0x0695), rows1 = kit.r8(0x0699), boundsX1 = kit.r16(0x0694), boundsY1 = kit.r16(0x0698),
    size069A = kit.r16(0x069A), size069C = kit.r16(0x069C), scrollMode06EF = kit.r16(0x06EF),
    low = low, hi = hi, first = first,
    playerX = kit.r16(A.playerX), playerY = kit.r16(A.playerY), mask = kit.r16(A.inputMask), banner = kit.r8(A.bannerFlag),
  }
end

-- OQ-3 data-flow check (scene_script BuildMetatileFlagTable + FillCollisionFromLayer0): the collision byte of cell i is
-- flagTable[mapData[i]] with the flag table in bank $7F (pointer $06AA) and the map data (one metatile index byte per cell,
-- bank $7E, start = word $069E). Counts the cells whose low nibble differs from the flag table byte (a high nibble is a
-- dynamic mark and is ignored), the words of the 256 metatile definitions (bank $7E, pointer $06AE, 4 words each) that still
-- have bit 9 set (the routine clears it), and returns the 4 words of the first metatile whose flag is each of `want`.
function gaia.coll.verifyFlow(want)
  want = want or { 6, 9 }
  local baseA, ftab, defp = kit.r16(0x069E), kit.r16(0x06AA), kit.r16(0x06AE)
  local n = kit.r8(0x0693) * kit.r8(0x0697) * 256
  local mism, firstm, used = 0, {}, {}
  for m = 0, 255 do used[m] = 0 end
  for i = 0, n - 1 do
    local m = kit.r8(baseA + i)
    used[m] = used[m] + 1
    local f = kit.r8(0x10000 + ftab + m) & 15
    local c = kit.r8(0x1C000 + i) & 15
    if c ~= f then
      mism = mism + 1
      if #firstm < 5 then firstm[#firstm + 1] = { i, m, c, f } end
    end
  end
  local bit9, merge, hist = 0, {}, {}
  for i = 1, 16 do hist[i] = 0 end
  local meta = {}
  for m = 0, 255 do
    local f = kit.r8(0x10000 + ftab + m) & 15
    if used[m] > 0 then hist[f + 1] = hist[f + 1] + used[m] end
    local words = {}
    for w = 0, 3 do
      local v = kit.r16(defp + m * 8 + w * 2)
      words[#words + 1] = v
      if used[m] > 0 and (v & 0x0200) ~= 0 then bit9 = bit9 + 1 end
    end
    for _, t in ipairs(want) do
      if f == t and used[m] > 0 and not meta[string.format("%02X", t)] then
        meta[string.format("%02X", t)] = { metatile = m, cellsUsingIt = used[m], words = words }
      end
    end
  end
  return { baseA = baseA, flagTable = ftab, defTable = defp, cells = n, mismatches = mism, firstMismatches = firstm,
           usedMetatileWordsWithBit9 = bit9, cellsPerFlagValue = hist, firstMetatileOfType = meta,
           scrollMode06EF = kit.r16(0x06EF), mergeWord06A2 = kit.r16(0x06A2) }
end

-- Runs gaia.coll.census at the `after`-th frame end (default 3) with a positive control: the number of writes of the
-- frame counter $36 seen up to then (the NMI writes it once per frame). Ends the run with the row (kit.finish).
function gaia.coll.censusRun(firsts, maxFirst, after)
  local ctl, f = 0, 0
  emu.addMemoryCallback(function(a, v)
    if (a & 0xFFFF) == 0x36 then ctl = ctl + 1 end
  end, emu.callbackType.write, 0x36, 0x36, emu.cpuType.snes, WR)
  kit.onFrame(function()
    f = f + 1
    if f < (after or 3) then return end
    local row = gaia.coll.census(firsts, maxFirst)
    row.control36Writes, row.frames = ctl, f
    kit.finish(row)
  end)
end

-- W1 census: who reads / writes an address ------------------------------------------------------
-- The PC reported inside a memory callback is the address AFTER the instruction that made the access.
-- gaia.watch{ {name=, lo=, hi=, kinds={"read","write"}, bits=mask?}, ... } logs each unique (name, kind, K:PC) once
-- (first `perKey` hits are counted, later ones only counted), and returns a table of counts for kit.finish.
local census, censusN = {}, 0
function gaia.pc()
  local st = emu.getState()
  return st["cpu.k"], st["cpu.pc"], st["frameCount"]
end

function gaia.once(tag, extra)
  local k, pc, f = gaia.pc()
  local key = string.format("%s %02X:%04X", tag, k, pc)
  local e = census[key]
  if not e then
    e = { n = 0, first = f, k = k, pc = pc }
    census[key] = e
    censusN = censusN + 1
    gaia.log("%s f=%d scene=%04X %s", key, f, kit.r16(gaia.addr.sceneCur), extra or "")
  end
  e.n = e.n + 1
  return e
end

function gaia.watch(list)
  for _, w in ipairs(list) do
    for _, kind in ipairs(w.kinds or { "read", "write" }) do
      emu.addMemoryCallback(function(a, v)
        if w.bits and kind == "write" then
          local old = kit.r8(a)
          if ((old ~ v) & w.bits) == 0 then return end
        end
        gaia.once(w.name .. " " .. kind, string.format("addr=%04X val=%02X", a, v))
      end, emu.callbackType[kind], w.lo, w.hi, emu.cpuType.snes, WR)
    end
  end
end

-- rows {key, n, first} sorted by key, for kit.finish
function gaia.censusRows()
  local rows = {}
  for key, e in pairs(census) do rows[#rows + 1] = { key = key, n = e.n, first = e.first } end
  table.sort(rows, function(a, b) return a.key < b.key end)
  return rows
end

-- script flags $0A00-$0AB3 (FR00) --------------------------------------------------------------------
-- Global flag index n = WRAM byte $0A00 + (n >> 3), bit (n & 7), n in 0..1439 ($0000-$059F); the dump covers the
-- 180 bytes $0A00-$0AB3 (flag-analysis-plan 2.1). Writes go through emu.write (they do not run memory callbacks:
-- measured in FR00, so a forced write never shows up in gaia.flagwatch).
--   gaia.flags.loc(n)          -> byte offset, bit
--   gaia.flags.get(n)          -> 0 or 1
--   gaia.flags.set(n) / clear(n)  -> the bit read back after the write (1 / 0)
--   gaia.flags.dump()          -> 360 hex digits (180 bytes of $0A00-$0AB3)
--   gaia.flags.diff(a, b)      -> list of changed flag indices (ascending) and a parallel list of "<index hex>:<old>><new>"; a, b are dumps
--   gaia.flagwatch(opts)       -> handle with :summary() and :stop(); write callback over the flag array
gaia.flags = { base = 0x0A00, bytes = 180 }

function gaia.flags.loc(n)
  assert(n >= 0 and n < gaia.flags.bytes * 8, "flag index out of range: " .. tostring(n))
  return gaia.flags.base + (n >> 3), n & 7
end

function gaia.flags.get(n)
  local a, b = gaia.flags.loc(n)
  return (kit.r8(a) >> b) & 1
end

function gaia.flags.set(n)
  local a, b = gaia.flags.loc(n)
  kit.w8(a, kit.r8(a) | (1 << b))
  return gaia.flags.get(n)
end

function gaia.flags.clear(n)
  local a, b = gaia.flags.loc(n)
  kit.w8(a, kit.r8(a) & (~(1 << b) & 0xFF))
  return gaia.flags.get(n)
end

function gaia.flags.dump()
  local t = {}
  for i = 0, gaia.flags.bytes - 1 do t[#t + 1] = string.format("%02X", kit.r8(gaia.flags.base + i)) end
  return table.concat(t)
end

function gaia.flags.diff(a, b)
  local out, detail = {}, {}
  for i = 0, gaia.flags.bytes - 1 do
    local x, y = tonumber(a:sub(i * 2 + 1, i * 2 + 2), 16), tonumber(b:sub(i * 2 + 1, i * 2 + 2), 16)
    local d = x ~ y
    if d ~= 0 then
      for bit = 0, 7 do
        if (d >> bit) & 1 == 1 then
          local n = i * 8 + bit
          out[#out + 1] = n
          detail[#detail + 1] = string.format("%X:%d>%d", n, (x >> bit) & 1, (y >> bit) & 1)
        end
      end
    end
  end
  return out, detail
end

-- Length of the memory-writing 65816 instructions (fixed length whatever M and X are): the instruction that ENDS at the
-- PC a callback reports is found by trying the starts pc-2, pc-3, pc-4 (and pc-1 never writes memory).
local WLEN = {}
for _, op in ipairs { 0x85, 0x95, 0x81, 0x91, 0x92, 0x87, 0x97, 0x83, 0x93, 0x86, 0x96, 0x84, 0x94, 0x64, 0x74, 0x04, 0x14,
                      0x06, 0x16, 0x46, 0x56, 0x26, 0x36, 0x66, 0x76, 0xE6, 0xF6, 0xC6, 0xD6 } do WLEN[op] = 2 end
for _, op in ipairs { 0x8D, 0x9D, 0x99, 0x8E, 0x8C, 0x9C, 0x9E, 0x0C, 0x1C, 0x0E, 0x1E, 0x4E, 0x5E, 0x2E, 0x3E, 0x6E, 0x7E,
                      0xEE, 0xFE, 0xCE, 0xDE, 0x54, 0x44 } do WLEN[op] = 3 end
for _, op in ipairs { 0x8F, 0x9F } do WLEN[op] = 4 end

-- ROM file offset of a CPU address of the HiROM map (banks $00-$3F and $80-$BF from $8000, banks $C0-$FF), or nil (RAM code).
function gaia.romOffset(k, pc)
  if k >= 0xC0 or ((k & 0x7F) <= 0x3F and pc >= 0x8000) then return ((k & 0x3F) << 16) | pc end
  return nil
end

-- Offset(s) of the instruction that ends at CPU K:PC, as a list (several when the bytes before the PC read as more than one
-- writer of different length; the harness settles it with the `; {n}` tags of extract:lt).
function gaia.insEndingAt(k, pc)
  local endOff = gaia.romOffset(k, pc)
  if not endOff then return nil, {} end
  local c = {}
  for _, len in ipairs { 3, 2, 4 } do
    local s = endOff - len
    if s >= 0 and WLEN[emu.read(s, emu.memType.snesPrgRom)] == len then c[#c + 1] = s end
  end
  return endOff, c
end

-- gaia.flagwatch{ lo=0x0A00, hi=0x0AB3, cap=200, keep=300, filterClear=true, clearEnd=253294, control=true }
-- Write callback over the flag array. A write that changes a bit logs one row (cap rows, then only counted):
--   f=<frame> sc=<sceneCurrent> <K>:<PC> end=<offset after the instruction> ins=<offset(s) of the instruction> $<addr> <old>><new> +<set flags> -<cleared flags>
-- (a 16-bit store gives two rows, one per byte). A write that stores the byte it already holds is counted as `noop`.
-- The zeroing of $0A00/$0A01 by ClearSceneState (the store that ends at `clearEnd`, 253294 on the original ROM:
-- STX $eventFlags at 253291, 3 bytes) is not logged per write: summary().clear has its count, frames and nonzero old values.
-- `control` also counts the NMI's INC $36 writes (summary().control36Writes, the standing positive control).
-- `actor` (FR10): append " D=<cpu D> entry=<bank>:<word at D, D+2> P=<$09A2>,<$09A4>" to every row (D = the actor slot id inside a COP handler, X is the bit index there; the word is the slot's current entry address).
-- `log` (FR10): a function(fmt, ...) that replaces gaia.log for the rows and the summary lines (a long play session uses its own
-- cap per segment).
function gaia.flagwatch(opts)
  opts = opts or {}
  local LOG = opts.log or gaia.log
  local lo, hi = opts.lo or gaia.flags.base, opts.hi or (gaia.flags.base + gaia.flags.bytes - 1)
  local cap, keep = opts.cap or 200, opts.keep or 300
  local filterClear = opts.filterClear ~= false
  local clearEnd = opts.clearEnd or 253294
  local fw = { logged = 0, noop = 0, hits = 0, overflow = 0, ctl = 0, rows = {}, byIns = {},
               clear = { n = 0, frames = {}, nonzeroOld = {} }, refs = {} }

  local ref = emu.addMemoryCallback(function(a, v)
    local off = kit.wramOffset(a) or (a & 0xFFFF)
    if off < lo or off > hi then return end
    fw.hits = fw.hits + 1
    local old = kit.r8(off)
    if old == v then fw.noop = fw.noop + 1; return end
    local k, pc, frame = gaia.pc()
    local endOff, cands = gaia.insEndingAt(k, pc)
    local scene = kit.r16(gaia.addr.sceneCur)
    if filterClear and endOff == clearEnd and (off == 0x0A00 or off == 0x0A01) then
      fw.clear.n = fw.clear.n + 1
      if fw.clear.frames[#fw.clear.frames] ~= frame and #fw.clear.frames < 20 then fw.clear.frames[#fw.clear.frames + 1] = frame end
      if old ~= 0 and #fw.clear.nonzeroOld < 20 then
        fw.clear.nonzeroOld[#fw.clear.nonzeroOld + 1] = string.format("f=%d $%04X %02X", frame, off, old)
      end
      return
    end
    local up, dn = {}, {}
    local base = (off - gaia.flags.base) * 8
    for bit = 0, 7 do
      if ((old ~ v) >> bit) & 1 == 1 then
        if (v >> bit) & 1 == 1 then up[#up + 1] = base + bit else dn[#dn + 1] = base + bit end
      end
    end
    local ins = {}
    for _, s in ipairs(cands) do ins[#ins + 1] = tostring(s) end
    local insText = #ins > 0 and table.concat(ins, "|") or "?"
    local key = string.format("%s end=%s", insText, tostring(endOff))
    local e = fw.byIns[key]
    if not e then e = { n = 0, first = frame }; fw.byIns[key] = e end
    e.n = e.n + 1
    local function fmt(list, sign)
      local t = {}
      for i = 1, math.min(#list, 8) do t[#t + 1] = string.format("%s%X", sign, list[i]) end
      if #list > 8 then t[#t + 1] = sign .. "..(" .. #list .. ")" end
      return table.concat(t, " ")
    end
    if fw.logged < cap then
      fw.logged = fw.logged + 1
      local who = ""
      if opts.actor then
        local d = emu.getState()["cpu.d"] or 0
        who = string.format(" D=%04X entry=%02X:%04X P=%d,%d", d, kit.r8(d + 2), kit.r16(d), kit.r16(0x09A2), kit.r16(0x09A4))
      end
      LOG("f=%d sc=%02X %02X:%04X end=%s ins=%s $%04X %02X>%02X %s %s%s", frame, scene, k, pc, tostring(endOff), insText, off, old, v,
               fmt(up, "+"), fmt(dn, "-"), who)
    else
      fw.overflow = fw.overflow + 1
    end
    if #fw.rows < keep then
      fw.rows[#fw.rows + 1] = { f = frame, sc = scene, k = k, pc = pc, ["end"] = endOff, ins = insText, a = off, old = old, new = v, set = up, clr = dn }
    end
  end, emu.callbackType.write, lo, hi, emu.cpuType.snes, WR)
  fw.refs[#fw.refs + 1] = { ref, emu.callbackType.write, lo, hi }

  if opts.control ~= false then
    local cref = emu.addMemoryCallback(function(a, v)
      if (a & 0xFFFF) == 0x36 then fw.ctl = fw.ctl + 1 end
    end, emu.callbackType.write, 0x36, 0x36, emu.cpuType.snes, WR)
    fw.refs[#fw.refs + 1] = { cref, emu.callbackType.write, 0x36, 0x36 }
  end

  -- Also logs the one summary line of the filtered ClearSceneState zeroing (when there was any) and of the rows past the cap.
  function fw:summary()
    local by = {}
    for key, e in pairs(self.byIns) do by[#by + 1] = { key = key, n = e.n, first = e.first } end
    table.sort(by, function(x, y) return x.key < y.key end)
    if self.clear.n > 0 then
      LOG("ClearSceneState zeroing of $0A00-$0A01 (store ending at %d): %d byte writes at frames %s; nonzero old values: %s",
               clearEnd, self.clear.n, table.concat(self.clear.frames, ","), #self.clear.nonzeroOld > 0 and table.concat(self.clear.nonzeroOld, "; ") or "none")
    end
    if self.overflow > 0 then LOG("flagwatch: %d further changing writes not logged (cap %d)", self.overflow, cap) end
    return { control36Writes = self.ctl, hits = self.hits, noop = self.noop, logged = self.logged, overflow = self.overflow,
             clear = self.clear, byIns = by, rows = self.rows }
  end

  function fw:stop()
    for _, r in ipairs(self.refs) do emu.removeMemoryCallback(r[1], r[2], r[3], r[4], emu.cpuType.snes, WR) end
    self.refs = {}
  end
  return fw
end

-- gaia.execMarks{ name = <ROM file offset of an instruction>, ... [, cap] } counts the executions of each instruction and keeps the
-- emulator frame of the first `cap` (10) of them: h.n[name], h.frames[name] (the frame count of emu.getState()). ROM-specific offsets.
function gaia.execMarks(map, cap)
  local h = { frames = {}, n = {} }
  for name, off in pairs(map) do
    h.frames[name], h.n[name] = {}, 0
    emu.addMemoryCallback(function()
      h.n[name] = h.n[name] + 1
      if #h.frames[name] < (cap or 10) then h.frames[name][#h.frames[name] + 1] = emu.getState().frameCount end
    end, emu.callbackType.exec, off, off, emu.cpuType.snes, emu.memType.snesPrgRom)
  end
  return h
end

-- per-frame trial ------------------------------------------------------------------------------
-- setup runs at the first end of frame (paint tiles, place the player); the buttons are then held for `frames`
-- frames and sample() is stored after each of them. The result {samples, final} is returned by run_script.
function gaia.trial(o)
  local n, samples, started = 0, {}, false
  local handle
  handle = kit.onFrame(function()
    if not started then
      started = true
      if o.setup then o.setup() end
      gaia.hold(o.hold)
      return
    end
    n = n + 1
    samples[#samples + 1] = o.sample and o.sample() or gaia.player()
    if n >= o.frames then
      kit.off(handle)
      gaia.release()
      local res = { samples = samples }
      if o.final then res.final = o.final() end
      kit.finish(res)
    end
  end)
end
```

```lua
-- debug (not a leg): flood-fill the collision layer from the player cell and print the map ('o' = reachable with the pass set, '.' = free-looking but unreachable, others raw low nibble, '*' = high nibble set)
local PASS = { [0] = true, [3] = true, [5] = true, [6] = true, [9] = true }
local s = fr10.snap()
local stride, rows = kit.r8(0x0693), kit.r8(0x0697)
local w, h = stride * 16, rows * 16
emu.log(string.format("scene=%02X cells=%dx%d p09A2=%d,%d", s.sc, w, h, s.x, s.y))
local function free(x, y) if x < 0 or y < 0 or x >= w or y >= h then return false end local v = gaia.coll.get(x, y); return PASS[v & 15] and (v >> 4) == 0 end
local seen, q, qi = {}, { { s.x >> 4, s.y >> 4 } }, 1
seen[(s.y >> 4) * w + (s.x >> 4)] = true
while qi <= #q do
  local c = q[qi]; qi = qi + 1
  for _, d in ipairs { {1,0}, {-1,0}, {0,1}, {0,-1} } do
    local nx, ny = c[1] + d[1], c[2] + d[2]
    if free(nx, ny) and not seen[ny * w + nx] then seen[ny * w + nx] = true; q[#q + 1] = { nx, ny } end
  end
end
local maxRow = math.min(h - 1, 63)
for ty = 0, maxRow do
  local t = {}
  for tx = 0, math.min(w - 1, 63) do
    local v = gaia.coll.get(tx, ty)
    local c = string.format("%X", v & 15)
    if (v >> 4) ~= 0 then c = "*" elseif seen[ty * w + tx] then c = "o" elseif free(tx, ty) then c = "." end
    t[#t + 1] = c
  end
  emu.log(string.format("%02d %s", ty, table.concat(t)))
end
kit.finish({ ok = true })
```
