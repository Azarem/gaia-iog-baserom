## M1_arrange_R.txt (final run: exec counters of the Arrange labels on R)

```lua
m1.run(function(m)
  local pick=m.exec(0x02B5BE,'ArrangePickTarget R')
  local tloop=m.exec(0x02B5E7,'ArrangeTargetLoop R')
  local swap=m.exec(0x02B615,'ArrangePerformSwap R')
  local sloop=m.exec(0x02B590,'ArrangeSourceLoop R')
  m.wait(120)
  local st=emu.getState()
  m.log('H1: cpu.k=%s cpu.sp=%s scene=%04X', tostring(st['cpu.k']), tostring(st['cpu.sp']), m.r16(0x0644))
  local function inv() local t={}; for i=0,3 do t[#t+1]=string.format('%02X', m.r8(0x0AB4+i)) end; return table.concat(t,' ') end
  local function setInv() for i=0,15 do m.w8(0x0AB4+i,0) end; m.w8(0x0AB4,0x06); m.w8(0x0AB5,0x01); m.w8(0x0AB6,0x0A); m.w16(0x0AC4,0xFFFF) end
  local function cnt() return string.format('srcLoop=%d pick=%d tgtLoop=%d swap=%d', sloop.n, pick.n, tloop.n, swap.n) end
  local function hold(btn, n)
    gaia.hold({btn}); m.wait(1)
    local jc=m.r16(0x0656)
    m.wait((n or 3)-1); gaia.release(); m.wait(1)
    return jc
  end
  local function trial(btn)
    setInv(); m.wait(3)
    local n=0
    while not m.inInventory() and n<4 do m.press({'select'},6); m.wait(40); n=n+1 end
    m.wait(30)
    hold('down'); m.wait(20)
    local c1=hold('a'); m.wait(30)
    m.log('%s: tab=%d after tab entry: %s (joypadCurrent while A held=%04X)', btn, m.r16(0x0AFA), cnt(), c1)
    local c2=hold('a'); m.wait(30)
    m.log('  after source A: %s (joypadCurrent=%04X)', cnt(), c2)
    hold('right'); m.wait(20)
    local s0=swap.n
    local c3=hold(btn); m.wait(40)
    m.log('  target phase press physical %s: joypadCurrent=%04X swapExec +%d, inv=%s', btn, c3, swap.n-s0, inv())
    n=0
    while m.inInventory() and n<8 do hold('b'); m.wait(30); n=n+1 end
    m.wait(20)
  end
  for _,b in ipairs({'x','y','a','b'}) do trial(b) end
end, {out='M1_arrange_R.txt'})
```
