# R2 scene sweep: Lua of ROM O / ROM R runs (2026-11-09)

Companion of `R2_20261109.md`. Every Lua text run in the pass is below, verbatim. The kits are in `kit/lib/` (`gaia_kit`, `mesen_kit`, `iog_kit` are the shared kits of earlier steps, not repeated); the Python drivers that fill the placeholders (`SID`, `GEMS`, the `pcs` table) are `harness/r2_sweep.py`, `r2_warp.py`, `r2_fw.py`, `r2_probes.py`, and `r2_static.py` for the static decode (no Lua).

Call forms (from the drivers):

- sweep: `r2.run{scene=<id>, post=180, maxWait=700 (250 with --flagpoke), pcs={...}, [flagPoke=65535], [noBanner=true for $FE/$FF], [coll=false]}` with `includes = ["gaia_kit", "r2_kit"]`, one `run_script` per scene after `load_state_file` of `orig_S01r_south-cape-ready.mss` (ROM O) or `R-0cfcb558_S01r_south-cape-ready.mss` (ROM R).
- warp probe: `r2w.run{scene=<src>, x=<rx*16+8>, y=<ry*16+16>, dest=<dest>, cwr=174459}` with `includes = ["gaia_kit", "r2w_kit"]`.
- forced-walk probe: `r2w.fw{scene=<src>, x=, y=, hooks={south=60334, north=60418, west=60502}, observe=300, cwr=174459}`.

## kit/lib/r2_kit.lua

```lua
-- description: R2 scene sweep: one scene per run (warp by sceneNext, watch, one result row) with the coverage hooks of the R2 questions.
-- requires: mesen_kit, iog_kit, gaia_kit
--
--   r2.run{ scene=, post=180, maxWait=700, pcs={...ROM offsets...}, flagPoke=nil, noBanner=false, coll=true }
--
-- Run it from a loaded state (S01r) with `run_script includes={"gaia_kit"}`. At the first frame end it (optionally) writes the word
-- `flagPoke` into $0A00 (event flags $00-$0F) and then `scene` into sceneNext $0642; it watches until `post` NMIs after the arrival
-- (sceneCurrent == scene and sceneNext == 0), or until `maxWait` NMIs without an arrival, then ends with one result row (kit.finish).
-- A sample is also taken 180 NMIs after the write (the literal "advance 180 frames" of the R2 prompt).
--
-- pcs (ROM file offsets of the loaded ROM; `false` or nil disables a hook):
--   nmi (NmiHandler)  cfg (the ASL after the operand read of SceneCmd_ConfigDisplay: A = preset index, Y = stream offset + 1)
--   m3 (SceneCmd_LoadBgTiles, mode 3 entry)  l4 (Load4bppPage)  fgj (JMP DmaLowVramTileset in SceneCmd_LoadLayerMaps)  lowv (DmaLowVramTileset)
--   fcs (FindCurrentScene)  ldpg (LoadPlayerGraphics)  par (ParallaxThinkerEntry)  parw (the word of ParallaxThinkerEntry, 16 bit)
--   copspawn (STA $0000,X in SpawnThinkerParam: A = entry word)  tefidx (instruction after the STA of TestEventFlag: $0000 = flag index)
r2 = r2 or {}

local PRG = emu.memType.snesPrgRom
local WR = emu.memType.snesWorkRam
local CPU = emu.cpuType.snes

local function r16(a) return emu.read16(a, WR, false) end
local function r8(a) return emu.read(a, WR, false) end
local function hx(v, w) return string.format("%0" .. (w or 2) .. "X", v) end

local function cap(t, v, n) if #t < (n or 8) then t[#t + 1] = v end end

function r2.run(o)
  local P = o.pcs or {}
  local post, maxWait = o.post or 180, o.maxWait or 700
  local S = {
    scene = o.scene, base = r16(0x0644), pre0642 = r16(0x0642), flagPoke = o.flagPoke,
    presets = {}, presetOperandOk = {}, par = {}, parCop = 0, ffTests = { T = 0, A = 0, O = 0 }, ffActorD = {},
    byte1FTests = 0, byte1FReads = 0, d58 = {}, d58n = 0, seen = { r16(0x0644) }, cov = {},
  }
  local nmiN, w36, nf = 0, 0, 0
  local started, start36 = false, 0
  local arrived, arriveN, arriveF = false, nil, nil
  local nmiAtArrive = 0
  local sample180 = nil

  -- positive controls
  if P.nmi then
    emu.addMemoryCallback(function()
      nmiN = nmiN + 1
      if arrived and not S.loadFrames then S.loadFrames = nf - arriveF end
    end, emu.callbackType.exec, P.nmi, P.nmi, CPU, PRG)
  end
  emu.addMemoryCallback(function(a) if (a & 0xFFFF) == 0x36 then w36 = w36 + 1 end end, emu.callbackType.write, 0x36, 0x36, CPU, WR)

  -- coverage counters (exec)
  for name, off in pairs({ m3 = P.m3, l4 = P.l4, fgj = P.fgj, lowv = P.lowv, fcs = P.fcs, ldpg = P.ldpg }) do
    if off then
      S.cov[name] = 0
      emu.addMemoryCallback(function() S.cov[name] = S.cov[name] + 1 end, emu.callbackType.exec, off, off, CPU, PRG)
    end
  end

  -- ConfigDisplay: preset index = A at the ASL; the operand byte read from the script stream must equal it
  if P.cfg then
    emu.addMemoryCallback(function()
      local st = emu.getState()
      local a, y = st["cpu.a"] & 0xFF, st["cpu.y"]
      local ptr = r16(0x3A) | (r8(0x3C) << 16)
      local bank = ptr >> 16
      local lo = (ptr & 0xFFFF) + y - 1
      local opv
      if bank == 0x7E or bank == 0x7F then
        opv = emu.read(((bank - 0x7E) << 16) + lo, WR, false)
      else
        opv = emu.read(((bank & 0x3F) << 16) + lo, PRG, false)
      end
      cap(S.presets, hx(a)); cap(S.presetOperandOk, opv == a and 1 or 0)
    end, emu.callbackType.exec, P.cfg, P.cfg, CPU, PRG)
  end

  -- parallax thinker: entry (spawn parameter = animScratch+2,X) and COP spawns that name the entry
  if P.par then
    emu.addMemoryCallback(function()
      local x = emu.getState()["cpu.x"]
      cap(S.par, hx(r8(0x10002 + x)), 16)
    end, emu.callbackType.exec, P.par, P.par, CPU, PRG)
  end
  if P.copspawn and P.parw then
    emu.addMemoryCallback(function()
      if (emu.getState()["cpu.a"] & 0xFFFF) == P.parw then S.parCop = S.parCop + 1 end
    end, emu.callbackType.exec, P.copspawn, P.copspawn, CPU, PRG)
  end

  -- TestEventFlag: flag index in $0000 (word) after the STA; byte $0A1F is flags $F8-$FF (idx >> 3 == $1F)
  if P.tefidx then
    emu.addMemoryCallback(function()
      local idx = r16(0x0000)
      S.tefAll = (S.tefAll or 0) + 1
      if (idx >> 3) == 0x1F then
        S.byte1FTests = S.byte1FTests + 1
        if idx == 0xFF then
          local d = emu.getState()["cpu.d"]
          if d >= 0x1000 then
            S.ffTests.A = S.ffTests.A + 1
            local k = hx(d, 4)
            S.ffActorD[k] = (S.ffActorD[k] or 0) + 1
          elseif d >= 0x0F00 and d < 0x1000 then S.ffTests.T = S.ffTests.T + 1
          else S.ffTests.O = S.ffTests.O + 1 end
        end
      end
    end, emu.callbackType.exec, P.tefidx, P.tefidx, CPU, PRG)
  end
  emu.addMemoryCallback(function(a) if (a & 0xFFFF) == 0x0A1F then S.byte1FReads = S.byte1FReads + 1 end end, emu.callbackType.read, 0x0A1F, 0x0A1F, CPU, WR)

  -- writes of the menu-choice words $0D58 / $0D5A: record the nonzero values with the writing pc
  emu.addMemoryCallback(function(a, v)
    if v ~= 0 then
      S.d58n = S.d58n + 1
      local st = emu.getState()
      cap(S.d58, string.format("%04X=%02X@%02X:%04X", a & 0xFFFF, v, st["cpu.k"], st["cpu.pc"]), 6)
    end
  end, emu.callbackType.write, 0x0D58, 0x0D5B, CPU, WR)

  local function sampleCpu()
    local st = emu.getState()
    return { k = st["cpu.k"], pc = st["cpu.pc"], sp = st["cpu.sp"], d = st["cpu.d"], dbr = st["cpu.dbr"], ps = st["cpu.ps"],
      bgMode = st["ppu.bgMode"], frame = st["frameCount"] }
  end

  local function finish(why)
    gaia.release()
    S.why = why
    S.arrived, S.arriveN, S.nmi, S.w36, S.frames = arrived, arriveN, nmiN, w36, nf
    S.sample180 = sample180
    S.endScene, S.endNext = r16(0x0644), r16(0x0642)
    S.bgm, S.actors, S.mask, S.banner = r8(0x06F2), r8(0x0DBC) | (r8(0x0DBD) << 8), r16(0x065A), r8(0x00B4)
    S.flagsEnd = r16(0x0A00)
    S.cpu = sampleCpu()
    S.layer = { l06EE = r16(0x06EE), l06EF = r16(0x06EF), tm = nil }
    S.visibleH = r16(0x06EC)
    S.tableIdx = r16(0x0646)
    if arrived and o.coll ~= false then
      local c = gaia.coll.census({}, 0)
      local h = {}
      for _, t in ipairs({ 3, 4, 5, 6, 9, 10, 12, 14 }) do h[hx(t)] = c.low[t + 1] end
      local dyn = 0
      for i = 2, 16 do dyn = dyn + c.hi[i] end
      S.coll = { w = c.cellsW, h = c.cellsH, stride = c.stride, bx = c.boundsX, by = c.boundsY, hist = h, dyn = dyn, clamped = c.clamped }
    end
    S.ps = nil
    kit.finish(S)
  end

  kit.onFrame(function()
    nf = nf + 1
    if not started then
      started = true
      start36 = r16(0x0036)
      if o.flagPoke then emu.write16(0x0A00, o.flagPoke, WR) end
      emu.write16(0x0642, o.scene, WR)
      return
    end
    local sc = r16(0x0644)
    if S.seen[#S.seen] ~= sc then cap(S.seen, sc, 10) end
    if not arrived and sc == o.scene and r16(0x0642) == 0 then
      arrived, arriveN, arriveF = true, nmiN, nf
      S.flagsArrive = r16(0x0A00)
      S.actorsArrive = r8(0x0DBC)
    end
    if r8(0x00B4) ~= 0 and not o.noBanner then gaia.hold((nf % 6 < 3) and { "a" } or {}) else gaia.release() end
    if not sample180 and nmiN >= 180 then
      sample180 = { scene = sc, next = r16(0x0642), arrived = arrived, bgm = r8(0x06F2), actors = r8(0x0DBC), flags = r16(0x0A00) }
    end
    if arrived and nmiN - arriveN >= post then
      if o.flagPoke then
        -- the flags are read once more 60 NMIs later by the caller's `post`; nothing extra here
      end
      finish("post")
    elseif nmiN >= maxWait or nf >= maxWait + 300 then
      finish("timeout")
    end
  end)
end
```

## kit/lib/r2w_kit.lua

```lua
-- description: R2 warp-entry probe: warp to a scene, put the player on a warp rectangle, let the warp fire, read the revive/save records.
-- requires: mesen_kit, iog_kit, gaia_kit
--
--   r2w.run{ scene=, x=, y=, dest=, settle=60, hold=90, maxWait=900 }
--
-- Run it from the S01r state. Frame 1 writes `scene` into sceneNext $0642. After `scene` has arrived ($0644 == scene, $0642 == 0) and $36
-- (frame parity counter, +1 per NMI) has advanced `settle` NMIs, the player actor position ($1014 / $1016, pixels) is written to (x, y) on every
-- frame (at most `hold` frames) until the warp stages a destination ($0642 != 0). The row (kit.finish) holds:
--   before / after = { ptr0AF0, bank0AF2, ptr0AF4, bank0AF6 } read at the staging frame (before) and after the destination arrived (after)
--   staged = $0642 at the staging frame, arrived = destination reached, pos = player (x, y) at staging, spawn = $064C / $064E / $0650 / $0652
r2w = r2w or {}

local WR = emu.memType.snesWorkRam
local function r16(a) return emu.read16(a, WR, false) end
local function r8(a) return emu.read(a, WR, false) end

function r2w.run(o)
  local settle, hold, maxWait = o.settle or 60, o.hold or 90, o.maxWait or 900
  local S = { scene = o.scene, dest = o.dest, tx = o.x, ty = o.y }
  local nf, started, arrivedAt36, staged, stagedF = 0, false, nil, nil, nil
  local arrivedDest = false
  local lastCwr, lastCwrF = 0, 0
  S.cwr, S.tile, S.pix = 0, 0, 0
  if o.cwr then
    local PRG, CPU = emu.memType.snesPrgRom, emu.cpuType.snes
    S.samp = {}
    emu.addMemoryCallback(function()
      S.cwr = S.cwr + 1
      if S.hf and #S.samp < 6 then
        local st = emu.getState()
        S.samp[#S.samp + 1] = { r8(0x09A6), r8(0x09A8), r16(0x09A2), r16(0x09A4), r16(0x00D4), r16(0x1014), r16(0x1016), st["cpu.a"], st["cpu.x"], st["cpu.d"], st["cpu.dbr"] }
      end
    end, emu.callbackType.exec, o.cwr, o.cwr, CPU, PRG)
    -- ConvertWarpToPixels entry: the tile AABB hit
    emu.addMemoryCallback(function() S.tile = S.tile + 1 end, emu.callbackType.exec, o.cwr + 166, o.cwr + 166, CPU, PRG)
    -- JMP ExecuteWarp: the pixel test passed
    emu.addMemoryCallback(function() S.pix = S.pix + 1 end, emu.callbackType.exec, o.cwr + 75, o.cwr + 75, CPU, PRG)
  end
  local function rec() return { r16(0x0AF0), r16(0x0AF2), r16(0x0AF4), r16(0x0AF6) } end
  local function finish(why)
    S.why = why
    S.after = rec()
    S.endScene = r16(0x0644)
    S.endNext = r16(0x0642)
    S.endPos = { r16(0x1014), r16(0x1016) }
    S.endFlags = { r16(0x1010), r16(0x09AE), r16(0x00D4), r16(0x00D6), r16(0x0B12), r16(0x09A2), r16(0x09A4), r16(0x09A6), r16(0x09A8) }
    kit.finish(S)
  end
  kit.onFrame(function()
    nf = nf + 1
    if not started then
      started = true
      S.start = rec()
      emu.write16(0x0642, o.scene, WR)
      return
    end
    local sc = r16(0x0644)
    -- a dialog or banner that blocks the main loop (no CheckWarpRectangles call for 20 frames) is dismissed with A pulses
    if S.cwr ~= lastCwr then lastCwr, lastCwrF = S.cwr, nf end
    if r8(0x00B4) ~= 0 or (o.cwr and nf - lastCwrF > 20) then
      gaia.hold((nf % 6 < 3) and { "a" } or {})
      S.apulse = (S.apulse or 0) + 1
    else
      gaia.release()
    end
    if not staged then
      if sc == o.scene and r16(0x0642) == 0 and not arrivedAt36 then arrivedAt36 = r16(0x0036) end
      if arrivedAt36 and ((r16(0x0036) - arrivedAt36) & 0xFFFF) >= settle then
        S.hf = (S.hf or 0) + 1
        emu.write16(0x1014, o.x, WR)
        emu.write16(0x1016, o.y, WR)
        if r16(0x0642) ~= 0 then
          staged, stagedF = r16(0x0642), nf
          S.staged = staged
          S.before = rec()
          S.pos = { r16(0x1014), r16(0x1016) }
          S.spawn = { r16(0x064C), r16(0x064E), r16(0x0650), r16(0x0652) }
        elseif S.hf > hold then
          finish("no-trigger")
        end
      end
    else
      if sc == staged and r16(0x0642) == 0 then
        arrivedDest = true
        S.arrived = true
        S.arriveF = nf - stagedF
        -- 5 more frames so the load finishes its record writes
        S.fin = (S.fin or 0) + 1
        if S.fin >= 5 then finish("arrived") end
      elseif nf - stagedF > maxWait then
        finish("timeout")
      end
    end
  end)
end

-- r2w.fw{ scene=, x=, y=, hooks={ south=<prg offset of the instruction after JSR ReadDirSprite_YVelocity>, north=..., ... },
--         settle=60, hold=600, observe=300, cwr= }
-- Forced-walk probe: same placement as r2w.run; the trigger is the first execution of one of the hook addresses (the instruction after the
-- first ReadDirSprite_* call of a ForcedWalk* actor). At the hook it records the entry word `$0000` (low byte = animation index, high byte =
-- move_chain_table index), the chain pointer the call loaded (`$2E` of the actor, `$2C` for east/west) and the actor slot (D). It then samples
-- every 10 frames for `observe` frames: player x / y ($1014 / $1016), animation index ($1028), joypadMaskStd $065A, playerFlags $09AE and the
-- camera scroll ($068A / $068E).
function r2w.fw(o)
  local settle, hold, observe = o.settle or 60, o.hold or 600, o.observe or 300
  local PRG, CPU = emu.memType.snesPrgRom, emu.cpuType.snes
  local S = { scene = o.scene, tx = o.x, ty = o.y, hits = {}, samples = {}, cwr = 0 }
  local nf, started, arrived36, trig, lastCwr, lastCwrF = 0, false, nil, nil, 0, 0
  for name, addr in pairs(o.hooks or {}) do
    emu.addMemoryCallback(function()
      local st = emu.getState()
      local d = st["cpu.d"]
      if #S.hits < 6 then
        S.hits[#S.hits + 1] = { name = name, nf = nf, d = d, w0 = r16(0x0000), c2e = r16((d + 0x2E) & 0xFFFF), c2c = r16((d + 0x2C) & 0xFFFF),
          stream = r16(0x0650), pan = r16(0x06E0) }
      end
      if not trig then trig = nf end
    end, emu.callbackType.exec, addr, addr, CPU, PRG)
  end
  if o.cwr then
    emu.addMemoryCallback(function() S.cwr = S.cwr + 1 end, emu.callbackType.exec, o.cwr, o.cwr, CPU, PRG)
    emu.addMemoryCallback(function() S.tile = (S.tile or 0) + 1 end, emu.callbackType.exec, o.cwr + 166, o.cwr + 166, CPU, PRG)
  end
  local function finish(why) S.why = why; S.endScene = r16(0x0644); S.endNext = r16(0x0642); kit.finish(S) end
  kit.onFrame(function()
    nf = nf + 1
    if not started then
      started = true
      emu.write16(0x0642, o.scene, WR)
      return
    end
    local sc = r16(0x0644)
    if S.cwr ~= lastCwr then lastCwr, lastCwrF = S.cwr, nf end
    if r8(0x00B4) ~= 0 or (o.cwr and not trig and nf - lastCwrF > 20) then gaia.hold((nf % 6 < 3) and { "a" } or {}) else gaia.release() end
    if not trig then
      if sc == o.scene and r16(0x0642) == 0 and not arrived36 then arrived36 = r16(0x0036) end
      if arrived36 and ((r16(0x0036) - arrived36) & 0xFFFF) >= settle then
        S.hf = (S.hf or 0) + 1
        emu.write16(0x1014, o.x, WR)
        emu.write16(0x1016, o.y, WR)
        if S.hf > hold then finish("no-trigger") end
      end
    else
      local k = nf - trig
      if k % 10 == 0 and #S.samples < 40 then
        S.samples[#S.samples + 1] = { k, r16(0x1014), r16(0x1016), r16(0x1028), r16(0x065A), r16(0x09AE), r16(0x068A), r16(0x068E), r16(0x06E2) }
      end
      if k > observe then finish("observed") end
    end
  end)
end
```

## F9 wait loop probe (`harness/r2_probes.py f9apu`)

```lua
-- O ROM, scene $F9 written into sceneNext at the first frame end; samples at frames 300 / 900 / 1500 of the run.
local WR, PRG, MEM = emu.memType.snesWorkRam, emu.memType.snesPrgRom, emu.memType.snesMemory
local nf, out = 0, { samples = {}, code = {} }
for i = 0, 15 do out.code[#out.code + 1] = emu.read(166828 + i, PRG, false) end   -- bytes at ROM 166828 (82:8BAC)
kit.onFrame(function()
  nf = nf + 1
  if nf == 1 then emu.write16(0x0642, 0xF9, WR) end
  if nf == 300 or nf == 900 or nf == 1500 then
    local st = emu.getState()
```

```lua
-- Positive control of the coverage hooks: exec counts at the entry of SceneCmd_LoadBgTiles (lbt) and at its mode-3 instruction (m3),
```

## Readers of `$0A1F` (`rd1f`, `SID` = $09 $1D $2A $5C $5D $60 $8A $CC $DE $FB $FD $FF)

```lua
-- Who reads $0A1F (flags $F8-$FF) in scene SID: read callback with the pc of the reader, plus TestEventFlag calls (index word at $0000
-- after the STA, ROM 45311) whose flag index is in $F8-$FF, with the direct page of the caller. 300 frames.
local WR, PRG, CPU = emu.memType.snesWorkRam, emu.memType.snesPrgRom, emu.cpuType.snes
local rd, tef = {}, {}
emu.addMemoryCallback(function(a)
  if (a & 0xFFFF) == 0x0A1F then
    local st = emu.getState()
    local k = string.format("%02X:%04X d=%04X", st["cpu.k"], st["cpu.pc"], st["cpu.d"])
    rd[k] = (rd[k] or 0) + 1
  end
end, emu.callbackType.read, 0x0A1F, 0x0A1F, CPU, WR)
emu.addMemoryCallback(function()
  local idx = emu.read16(0x0000, WR, false)
  if (idx >> 3) == 0x1F then
    local k = string.format("idx=%02X d=%04X", idx & 0xFF, emu.getState()["cpu.d"])
    tef[k] = (tef[k] or 0) + 1
  end
end, emu.callbackType.exec, 45311, 45311, CPU, PRG)
local nf = 0
kit.onFrame(function()
  nf = nf + 1
  if nf == 1 then emu.write16(0x0642, SID, WR) end
  if nf >= 300 then kit.finish({ reads = rd, tef = tef, sceneCurrent = emu.read16(0x0644, WR, false) }) end
```

## Game-over probe (`gameover`, `GEMS` = 0 and 100)

```lua
-- Game-over probe: set gemCount (GEMS) and playerHp = 0 (CheckPlayerDeath then starts GameOverSequence), pulse A while a banner blocks
-- the main loop, run to frame 1300 and report the scene reached, the transition words and the two warp records.
local WR = emu.memType.snesWorkRam
local function r16(a) return emu.read16(a, WR, false) end
local nf, S = 0, { seen = {}, gems = GEMS }
local last
kit.onFrame(function()
  nf = nf + 1
  if nf == 1 then
    S.before = { r16(0x0644), r16(0x0AF0), r16(0x0AF2), r16(0x0AF4), r16(0x0AF6), r16(0x0AD6), r16(0x0ACE) }
    emu.write16(0x0AD6, GEMS, WR)
    emu.write16(0x0ACE, 0, WR)
  end
  local sc = r16(0x0644)
  if sc ~= last then S.seen[#S.seen + 1] = { nf, sc }; last = sc end
  if emu.read(0x00B4, WR, false) ~= 0 then gaia.hold((nf % 6 < 3) and { "a" } or {}) else gaia.release() end
  if r16(0x0642) ~= 0 and not S.staged then S.staged = { nf, r16(0x0642), r16(0x064C), r16(0x064E), r16(0x0650), r16(0x0652) } end
  if nf >= 1300 then
    gaia.release()
    S.endScene, S.gemsEnd, S.hpEnd, S.wake = sc, r16(0x0AD6), r16(0x0ACE), r16(0x0AF8)
    kit.finish(S)
  end
end)
```
