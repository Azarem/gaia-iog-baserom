# E4 Lua (verbatim where inline, by file where saved)

All runs: `load_script` / `run_script` through the `user-mesen2-mcp-dev` namespace (server 0.6.0 started by `harness/mcp_client.py` with `--mcp-lib-dir=kit/lib` and `--mcp-results-dir=raw`), `turbo: true`, ROM O, each run preceded by its own `load_state_file` (`run_script` does not reload the state, T-51). The harness is `harness/e4_run.py` (commands `ident`, `build`, `sample`, `sweep`, `chain`), the viewer `harness/e4_view.py`, the screenshot differ `harness/e4_png.py` (stdlib PNG reader), the table builder `harness/e4_report.py`. The kit module `kit/lib/e4_kit.lua` (475 lines, includes `gaia_kit`, `e1_display`; the sweep also `e2_palette`) is saved as a file and is the verbatim Lua of every sampling, warp, chain and sweep run. Its entry points: `e4.install` (positive controls, PPU register callbacks `$2123-$2133` on all 128 bank aliases, scroll callbacks `$210D-$2114`, bind census for `ResetHdmaState` / `SetupHdmaChannel_Indirect` / `_Direct` / the `ADC #$10`, parallax PRE / POST accumulator dumps), `e4.sampler` (sample every N frames, `S.rec` set the frame before), `e4.warp`, `e4.chain`, `e4.bindSweep`.

## Dao window rectangle, first attempt (`harness/e4_dao_rect.lua`): frames 120, 123, 126 of one run with the `W12SEL` write forced to 0 from frame 120 (noisy: the scene animates)

```lua
-- E4 OQ-I035-4: Dao window rectangle. Player actor moved next to the rectangle (disposable state), A/B render: window 1 BG1 enable (W12SEL bit 1) on vs forced off by a write override.
local WR = emu.memType.snesWorkRam
local SM = emu.memType.snesMemory
local CPU = emu.cpuType.snes
local S = e4.install{ census = false }
local px, py = tonumber(DAO_PX) or 352, tonumber(DAO_PY) or 395
local override = nil
for bank = 0, 0xBF do
  if bank <= 0x3F or bank >= 0x80 then
    emu.addMemoryCallback(function() if override ~= nil then return override end end, emu.callbackType.write, (bank << 16) | 0x2123, (bank << 16) | 0x2123, CPU, SM)
  end
end
local function grab()
  -- whole frame in one call: buffer index = y * width + x + 1
  local buf = emu.getScreenBuffer()
  local t = {}
  for y = 0, 223 do
    local r = {}
    for x = 0, 255 do r[x + 1] = buf[y * 256 + x + 1] end
    t[y + 1] = r
  end
  return t
end
local n, A, B, C, linesA = 0, nil, nil, nil, nil
local camA
kit.onFrame(function()
  n = n + 1
  if n == 1 then
    emu.write16(0x1014, px, WR); emu.write16(0x1016, py, WR); emu.write16(0x09A2, px + 8, WR); emu.write16(0x09A4, py + 16, WR)
  end
  if n == 119 then S.rec = true; S.lines = {} end
  if n == 120 then
    S.rec = false
    A = grab()
    camA = string.format("bg1H=%04X mapV=%04X cameraTargetX=%04X cameraTargetY=%04X player=%04X,%04X", emu.read16(0x68A, WR, false), emu.read16(0x68E, WR, false), emu.read16(0x6BE, WR, false),
      emu.read16(0x6C2, WR, false), emu.read16(0x1014, WR, false), emu.read16(0x1016, WR, false))
    local r0, r1 = {}, {}
    for _, e in ipairs(S.lines[0x2126] or {}) do r0[#r0 + 1] = e end
    local l0, l1 = S.lines[0x2126] or {}, S.lines[0x2127] or {}
    local function runs(list)
      local out, cur = {}, nil
      for _, e in ipairs(list) do
        local sc, v = e >> 8, e & 0xFF
        if cur and cur.v == v and sc == cur.b + 1 then cur.b = sc else cur = { a = sc, b = sc, v = v }; out[#out + 1] = cur end
      end
      local s = {}
      for i, r in ipairs(out) do if i <= 20 then s[#s + 1] = string.format("%d-%d=%02X", r.a, r.b, r.v) end end
      return table.concat(s, " "), #out
    end
    local a, na = runs(l0)
    local b, nb = runs(l1)
    emu.log("WH0 runs(" .. na .. "): " .. a)
    emu.log("WH1 runs(" .. nb .. "): " .. b)
    emu.log(camA)
    override = 0x00
  end
  if n == 123 then B = grab() end
  if n == 126 then
    C = grab()
    local function diff(P, Q)
      local cnt, x0, x1, y0, y1 = 0, 999, -1, 999, -1
      for y = 1, 224 do for x = 1, 256 do
        if P[y][x] ~= Q[y][x] then cnt = cnt + 1; if x < x0 then x0 = x end; if x > x1 then x1 = x end; if y < y0 then y0 = y end; if y > y1 then y1 = y end end
      end end
      return string.format("diff=%d bbox x=%d-%d y=%d-%d", cnt, x0 - 1, x1 - 1, y0 - 1, y1 - 1)
    end
    local rowdiff = {}
    for y = 1, 224, 8 do
      local c, mn, mx = 0, 999, -1
      for x = 1, 256 do if A[y][x] ~= B[y][x] then c = c + 1; if x < mn then mn = x end; if x > mx then mx = x end end end
      rowdiff[#rowdiff + 1] = string.format("y=%d n=%d x=%d-%d", y - 1, c, mn - 1, mx - 1)
    end
    local function colDiff(P, Q, x)
      local c = 0
      for y = 1, 224 do if P[y][x + 1] ~= Q[y][x + 1] then c = c + 1 end end
      return c
    end
    local cols = {}
    for x = 0, 4 do cols[#cols + 1] = string.format("x=%d A-vs-B rows=%d B-vs-C rows=%d", x, colDiff(A, B, x), colDiff(B, C, x)) end
    kit.finish({ cols = cols, cam = camA, AvsB_windowOnVsOff = diff(A, B), BvsC_noise_windowOffBoth = diff(B, C), rows = rowdiff, w12 = emu.getState()["ppu.window[0].activeLayers[0]"] })
  end
end)
```

## Dao window rectangle, same-frame A/B (`harness/e4_dao_ss.lua`, run with `python e4_dbg.py orig_E4_dao.mss 240 e4_dao_ss.lua`)

```lua
-- E4 OQ-I035-4: Dao window rectangle, same-frame A/B inside one run with emu.createSavestate / emu.loadSavestate.
-- createSavestate / loadSavestate only work inside an exec callback of the main CPU: the NMI entry (ROM offset 33528) is used. After frame 120 a savestate S is taken at an NMI entry; the next completed frame is grabbed (A, game window); at the following NMI entry S is loaded back and the same frame is rendered again with the W12SEL write forced to 0 (B). Same state, same frame: the pixel difference is the window.
local WR = emu.memType.snesWorkRam
local SM = emu.memType.snesMemory
local CPU = emu.cpuType.snes
local ov = nil
for bank = 0, 0xBF do
  if bank <= 0x3F or bank >= 0x80 then
    emu.addMemoryCallback(function() return ov end, emu.callbackType.write, (bank << 16) | 0x2123, (bank << 16) | 0x2123, CPU, SM)
  end
end
local function grab()
  local buf = emu.getScreenBuffer()
  local t = {}
  for y = 0, 223 do for x = 0, 255 do t[y * 256 + x + 1] = buf[y * 256 + x + 1] end end
  return t
end
local n, ss, A, stage, after = 0, nil, nil, 0, 0
local function safe(f) return function(...) local ok, e = pcall(f, ...); if not ok then emu.log("E4 LUA ERROR " .. tostring(e)) end end end
emu.addMemoryCallback(safe(function()
  if stage == 0 and n >= 120 then
    local s = emu.createSavestate()
    ss = s; stage = 1; emu.log("savestate bytes " .. #ss .. " at frame " .. n)
  elseif stage == 2 then
    ov = 0
    emu.loadSavestate(ss)
    stage = 3; after = 0
    emu.log("loadSavestate at frame " .. n)
  end
end), emu.callbackType.exec, 33528, 33528, CPU, emu.memType.snesPrgRom)
kit.onFrame(safe(function()
  n = n + 1
  if n == 1 then
    emu.write16(0x1014, 352, WR); emu.write16(0x1016, 395, WR); emu.write16(0x09A2, 360, WR); emu.write16(0x09A4, 411, WR)
  end
  if stage == 1 then stage = 1.5
  elseif stage == 1.5 then A = grab(); stage = 2
  elseif stage == 3 then
    after = after + 1
    if after == 2 then
      local B = grab(); stage = 4
      local inside, outside, bbox = 0, 0, { 999, -1, 999, -1 }
      local ib, ob, samp, outl = { 999, -1, 999, -1 }, { 999, -1, 999, -1 }, {}, {}
      for y = 0, 223 do for x = 0, 255 do
        if A[y * 256 + x + 1] ~= B[y * 256 + x + 1] then
          if x >= 96 and x <= 159 and y >= 100 and y <= 147 then
            inside = inside + 1; if x < ib[1] then ib[1] = x end; if x > ib[2] then ib[2] = x end; if y < ib[3] then ib[3] = y end; if y > ib[4] then ib[4] = y end
            if #samp < 3 then samp[#samp + 1] = string.format('(%d,%d) A=%06X B=%06X', x, y, A[y * 256 + x + 1] & 0xFFFFFF, B[y * 256 + x + 1] & 0xFFFFFF) end
          else
            outside = outside + 1; if #outl < 18 then outl[#outl + 1] = string.format('%d,%d A=%06X B=%06X', x, y, A[y * 256 + x + 1] & 0xFFFFFF, B[y * 256 + x + 1] & 0xFFFFFF) end; if x < ob[1] then ob[1] = x end; if x > ob[2] then ob[2] = x end; if y < ob[3] then ob[3] = y end; if y > ob[4] then ob[4] = y end
          end
          if x < bbox[1] then bbox[1] = x end; if x > bbox[2] then bbox[2] = x end
          if y < bbox[3] then bbox[3] = y end; if y > bbox[4] then bbox[4] = y end
        end
      end end
      kit.finish({ inside_rect = inside, outside_rect = outside, inside_bbox = ib, outside_bbox = ob, samples = samp, outside_list = outl, n = n })
    end
  end
end))
```
