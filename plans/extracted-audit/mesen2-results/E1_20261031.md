## OQ-I066-3: Aura OBSEL

Two casts exist in the game and only one of them uses the spriteset the question is about. The first attempt (Freedan Aura Barrier) was prepared as the question text implies; the user then pointed out that the "Aura" item belongs to Shadow, so both were run.

Observations, Freedan Aura Barrier (`raw/E1_aura_barrier_freedan3_20261031.txt`; warp to scene `$3F`, form `$0AD4` = 1 (Freedan), abilities `$0AA2` = `$7777`, hold the attack button (physical B in ROM O) from t = 14, press L at t = 170):

- Compared byte for byte, VRAM `$4400-$46FF` (1536 bytes) equals the ROM bytes of `misc_fx_1CC480` (1536 / 1536 equal, 0 differences).
- `get_oam` after the cast: the orbital Barrier sprites use tiles `$61` and `$65`, palette 2, nameTable 0; PPU `oamBaseAddress` = `$4000` (name base 0 = word `$4000`), gap `$1000`, size mode 0 (8x8 / 16x16), that is OBSEL (`$2101`) bits 0-2 = 2.

--
-- run_script {includes={"gaia_kit","e1_cache"}, code='e1c.run{scenes={0x23,0x30}, settle=60, out="E1_oq1.txt"}'}
--
-- Callbacks run BEFORE the access (old value is still readable; the PC they report is the address after the writing instruction).
-- Offsets are decimal PRG (ROM file) offsets: off = ((bank & 0x3F) << 16) | pc16.
-- Each scene: wait until $0642 = 0, write the scene id to $0642 (quick warp), wait for $0644 = scene and $0642 = 0, run o.settle emulator frames.
-- The lines of a scene are appended to o.out when the scene's settle time ends (a crash in a later scene keeps the earlier lines);
-- a scene whose arrival does not happen within o.warpTimeout frames (default 600), or a wait for $0642 = 0 that lasts that long, ends the run with an error result.
-- o.append = true appends to o.out instead of replacing it (for several batches into one file).
-- Line format: s=<scene> f=<frame> <addr> <old>-><new> pc=<offset> win=<$0664>-<$0666> page=<$0668>
e1c = e1c or {}

local WR  = emu.memType.snesWorkRam
local CPU = emu.cpuType.snes
local function off24(k, pc) return ((k & 0x3F) << 16) | (pc & 0xFFFF) end

function e1c.run(o)
  local lines, nw, nz, byAddr, nlines = {}, 0, 0, {}, 0
  local cur, wroteAny = 0, o.append == true
  emu.addMemoryCallback(function(a, v)
    local addr = a & 0xFFFF
    local old = kit.r8(addr)
    local st = emu.getState()
    nw = nw + 1
    if v == 0 and old ~= 0 then nz = nz + 1 end
    local key = string.format("%04X", addr)
    byAddr[key] = (byAddr[key] or 0) + 1
    lines[#lines + 1] = string.format("s=%02X f=%d %04X %02X->%02X pc=%d win=%04X-%04X page=%04X", cur, st["frameCount"], addr, old, v,
      off24(st["cpu.k"], st["cpu.pc"]), kit.r16(0x0664), kit.r16(0x0666), kit.r16(0x0668))
  end, emu.callbackType.write, 0x066C, 0x0681, CPU, WR)

    if #lines > 0 then
      emu.mcp.writeFile(o.out, table.concat(lines, "\n") .. "\n", wroteAny)
      wroteAny = true; nlines = nlines + #lines; lines = {}
    end
  end

  local idx, stage, n, arriveN = 1, "wait", 0, 0
  kit.onFrame(function()
    n = n + 1
    local sc, sn = kit.r16(0x0644), kit.r16(0x0642)

      flush()
      kit.finish({ error = "wait_timeout", scene = cur, scene644 = sc, next = sn, writes = nw, lines = nlines, byAddr = byAddr })
      return
    end
    if stage == "wait" then
      if n >= 5 and sn == 0 then
        cur = o.scenes[idx]
        kit.w16(0x0642, cur)
        stage = "warping"
      end
    elseif stage == "warping" then
      if sc == cur and sn == 0 then stage = "run"; arriveN = n
      elseif n > (o.warpTimeout or 600) then
        flush()
        kit.finish({ error = "warp_timeout", scene = cur, scene644 = sc, next = sn, writes = nw, lines = nlines, byAddr = byAddr })
        return
      end
    elseif stage == "run" then

        flush()
        idx = idx + 1
        if idx > #o.scenes then
          kit.finish({ writes = nw, nonzeroToZero = nz, byAddr = byAddr, lines = nlines, scenes = #o.scenes })
        else
          stage = "wait"; n = 0
        end
      end
    end
  end)
end

    if stage == "wait" then
      if n >= 5 and sn == 0 then
        for _, p in ipairs(o.pokes or {}) do if p.w == 16 then kit.w16(p.addr, p.v) else kit.w8(p.addr, p.v) end end
        kit.w16(0x0642, o.scene)
        ev(string.format("f=%d STAGE scene %02X pokes=%d", f, o.scene, #(o.pokes or {})))
        stage = "warping"
      end
    elseif stage == "warping" then
      if sc == o.scene and sn == 0 then stage = "banner" end
      if n > (o.warpTimeout or 1500) then kit.finish({ error = "warp_timeout", scene = sc, next = sn, ev = S.ev }); return end
    end
    if stage == "banner" then
      if kit.r8(0x00B4) ~= 0 then
        bannerPulse = bannerPulse + 1
        gaia.hold((bannerPulse % 6 < 3) and { "a" } or {})
      else
        gaia.release(); stage = "run"; S.t = 0; S.arrivedFrame = f; S.n36Start = kit.r16(0x36)
        ev(string.format("f=%d RUN start (banner clear) scene=%02X n36=%d", f, sc, S.n36Start))
      end
    elseif stage == "run" then
      S.t = (kit.r16(0x36) - S.n36Start) & 0xFFFF
      local b = (S.t >= 1 and o.sched) and o.sched(S.t) or nil
      if b then gaia.hold(b) else gaia.release() end
      if o.tick then o.tick(S.t, S) end
      if snapAt[S.t] and S.snapT ~= S.t then
        S.snapT = S.t
        local s = e1.snap(); s.tag = "t" .. S.t
        local slot = kit.r16(0x09AA)
        S.snaps[#S.snaps + 1] = e1.fmt(s) .. string.format(" | form=%d func=%04X vram4400nz=%d adhoc=%02X%02X%02X/%02X%02X/%02X%02X $0AA2=%04X inv0=%02X eqIdx=%04X eqType=%04X plX=%d plY=%d sEw=%d sNs=%d pflags=%04X",
          kit.r16(0x0AD4), kit.r16(slot), vramOcc(),
          kit.r8(0x10C05), kit.r8(0x10C04), kit.r8(0x10C03), kit.r8(0x10C08), kit.r8(0x10C07), kit.r8(0x10C0A), kit.r8(0x10C09),
          kit.r16(0x0AA2), kit.r8(0x0AB4), kit.r16(0x0AC4), kit.r16(0x0AC6), kit.r16(0x1014), kit.r16(0x1016), kit.r16(0x09B2, nil, true), kit.r16(0x09B4, nil, true), kit.r16(0x09AE))
      end
      if S.t >= o.post then
        local lines = {}
        for _, l in ipairs(S.ev) do lines[#lines + 1] = l end
        for _, l in ipairs(S.snaps) do lines[#lines + 1] = l end
        if o.out then emu.mcp.writeFile(o.out, table.concat(lines, "\n") .. "\n", false) end
        local ex = {}
        for k, v in pairs(S.exec) do ex[k] = { n = v.n, first = v.first, t = v.t } end
        local shown = {}
        for i = 1, math.min(o.show or 14, #S.ev) do shown[i] = S.ev[i] end
        kit.finish({ exec = ex, obselWrites = obsel.n, adhocWrites = adhoc.n, events = shown, nEvents = #S.ev, snaps = S.snaps, scene = sc, frame = f, custom = S.custom })
      end
    end
  end)
end
