Frame `n` always counts from the frame of the state load (S08: frame 398; S01r: frame 377; S07: frame 1500).

## Fill probe (data words right after the fill loop; S08)

```lua
local WR = emu.memType.snesWorkRam
local PRG = emu.memType.snesPrgRom
local R = { w36 = 0, nmi = 0, hit = nil, n = 0 }
local base = { 0x7000, 0x7800, 0x8000 }
emu.addMemoryCallback(function(a) if (a & 0xFFFF) == 0x36 then R.w36 = R.w36 + 1 end end, emu.callbackType.write, 0x36, 0x36, emu.cpuType.snes, WR)
emu.addMemoryCallback(function() R.nmi = R.nmi + 1 end, emu.callbackType.exec, 33528, 33528, emu.cpuType.snes, PRG)
emu.addMemoryCallback(function()
  if R.hit then return end
  local row = { n = R.n }
  for t = 1, 3 do
    local nz, cnt, term = 0, 0, emu.read(base[t] + 224 * 3, WR, false)
    local sample = {}
    for i = 0, 223 do
      if emu.read(base[t] + i * 3, WR, false) == 1 then cnt = cnt + 1 end
      local d1 = emu.read(base[t] + i * 3 + 1, WR, false)
      local d2 = emu.read(base[t] + i * 3 + 2, WR, false)
      if d1 ~= 0 or d2 ~= 0 then nz = nz + 1 end
      if i < 6 then sample[#sample + 1] = string.format('%02X%02X', d2, d1) end
    end
    row[t] = string.format('tbl%d countBytes1=%d term=%02X dataWordsNonzero=%d first6=%s', t, cnt, term, nz, table.concat(sample, ','))
  end
  R.hit = row
end, emu.callbackType.exec, 240059, 240059, emu.cpuType.snes, PRG)
local c = 0
emu.addEventCallback(function() c = c + 1; R.n = c; if c >= 200 then emu.mcp.result({ w36 = R.w36, nmi = R.nmi, hit = R.hit }); emu.mcp.done() end end, emu.eventType.endFrame)
```

Result: `n` 161; table 1, 2, 3 data words nonzero 50, 54, 38; `w36` = `nmi` = 27.
