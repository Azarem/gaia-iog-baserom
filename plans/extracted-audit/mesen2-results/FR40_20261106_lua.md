# FR40 verbatim Lua (companion of FR40_20261106.md)

Every trial is one atomic `run_script` with `includes: ["fr40_kit"]` started by `harness/fr40.py` (`trial`), `fr40_n.py`, `fr40_g.py`, `fr40_g2.py`, `fr40_g3.py` (state `orig_S01r_south-cape-ready.mss` loaded, one frame advanced, then the script). The call lines are stored in the `lua` field of every row of `raw/FR40_rows.jsonl`. Forms used:

```lua
-- description: FR40 scene-entry sweep helpers: one scene arrival (quick warp, banner, watch window, optional flags) and the 20-frame direction test.
-- requires: fr20_kit
--
-- Load with run_script includes=["fr40_kit"] (the server resolves "requires": fr20_kit, gaia_kit, mesen_kit, iog_kit).
-- Nothing here writes the low direct page ($00-$94). Flag writes use emu.write (they do not fire memory callbacks).
--
--   fr40.arrive(P)   one scene arrival from the loaded state. P fields:
--        scene       scene id written to sceneNext ($0642) last, after the flags (nil: observe the loaded state)
--        set         global flag indices written before the warp (persistent flags; $0000-$000F would be cleared by the load)
--        watch       frames observed after the banner is gone (default 60)
--        input       "none" (default) or "mashA": physical A pressed 2 frames of every 12 during the watch window only
--        dest        { x, y, facing, scroll } written to warpDestX/Y, warpFacing, warpScrollParams before sceneNext (full warp; nil = quick warp)
--        stopFree    end the watch early when the player has been free (mask 0 and playerFlags $0800 clear) for that many frames
--        tag         copied into the result
--      result R: { tag, scene, error, arrivedF (kit frames from the start), n36 (frames of the watch), banner (frames the banner flag
--        $00B4 was nonzero), mask0/pf0/dmf0 (at the end of the banner), mask1/pf1/dmf1/sceneEnd/pos1 (at the end), releaseN (first watch
--        frame of a free run of 20 frames, or nil), tl (rows when mask, playerFlags $0800, dialogue mode $2000 or scene change), hist (scene changes with
--        frame), actors0/actors1 (compact "bank:pc@slot" lists), act0/act1 (live list length), cnt0DBC, hp1/hpMax1 (player HP $0ACE and maximum $0ACA at the end), bgm, bounds, cells0 (3x3 cells
--        around the player, low nibbles), flagsPre/flags0/flags1 (40 bytes of $0A00 as hex), ctl36 (writes of $36), frames, k/sp (CPU) }
--   fr40.move(P)     the direction test from the current state: hold = 20 frames per direction, gap = 8 frames between them,
--                    dirs = {"down","up","left","right"}; result { seg = { {dir, x0, y0, x1, y1, dx, dy, sc}, ... }, mask0, pf0, mask1, ... }
fr40 = fr40 or {}

local STX_EVENT_FLAGS = 253291   -- ClearSceneState's STX $eventFlags (O): one execution per scene load

local function actorsCompact()
  local out, s, guard = {}, kit.r16(0x56), 0
  while s >= 0x1000 and s < 0x1000 + 0x30 * 84 and guard < 100 do
    out[#out + 1] = string.format("%02X:%04X@%X", kit.r8(s + 2), kit.r16(s), s)
    s = kit.r16(s + 6)
    guard = guard + 1
  end
  return table.concat(out, ","), #out
end

-- low nibble of the collision cells around the player actor position (3 x 3, row-major), nil outside the map
local function cells3()
  local x, y = kit.r16(0x1014) - 8, kit.r16(0x1016) - 16   -- the special-tile probe point (CheckActorOnSpecialTile): actor position minus (8, 16)
  local bx, by = kit.r16(0x0692), kit.r16(0x0696)
  if x < 0 or y < 0 then return "--" end
  local tx, ty = x >> 4, y >> 4
  local t = {}
  for dy = -1, 1 do
    for dx = -1, 1 do
      local cx, cy = tx + dx, ty + dy
      if cx < 0 or cy < 0 or (cx << 4) >= bx or (cy << 4) >= by or bx == 0 or by == 0 then t[#t + 1] = "--"
      else t[#t + 1] = string.format("%02X", gaia.coll.get(cx, cy)) end
    end
  end
  return table.concat(t, " ")
end

local function cpuSane()
  local st = emu.getState()
  return st["cpu.k"], st["cpu.sp"]
end

function fr40.arrive(P)
  local R = { tag = P.tag, scene = P.scene, hist = {}, tl = {}, ctl36 = 0 }
  local watch = P.watch or 60
  local loadStarts = 0
  emu.addMemoryCallback(function(a, v) if (a & 0xFFFF) == 0x36 then R.ctl36 = R.ctl36 + 1 end end,
    emu.callbackType.write, 0x36, 0x36, emu.cpuType.snes, gaia.WR)
  emu.addMemoryCallback(function() loadStarts = loadStarts + 1 end,
    emu.callbackType.exec, STX_EVENT_FLAGS, STX_EVENT_FLAGS, emu.cpuType.snes, emu.memType.snesPrgRom)
  local lastSc = kit.r16(0x0644)
  local lastKey = ""
  local pulsing = false
  local freeRun = 0
  local n = 0
  local m
  local function tl(nn)
    local mask, pf, sc = kit.r16(0x065A), kit.r16(0x09AE), kit.r16(0x0644)
    local key = string.format("%04X %d %d %04X", mask, (pf >> 11) & 1, (kit.r16(0x09EC) >> 13) & 1, sc)
    if key ~= lastKey then
      lastKey = key
      if #R.tl < 40 then R.tl[#R.tl + 1] = { n = nn, f = kit.frame, mask = mask, pf = pf, sc = sc } end
    end
  end
  kit.onFrame(function()
    local sc = kit.r16(0x0644)
    if sc ~= lastSc then
      if #R.hist < 30 then R.hist[#R.hist + 1] = { f = kit.frame, from = lastSc, to = sc } end
      lastSc = sc
    end
  end)
  m = kit.machine{ start = "stage", states = {
    stage = function(mm)
      if mm:elapsed() < 2 then return end
      if P.scene and kit.r16(0x0642) ~= 0 then return end
      for _, f in ipairs(P.set or {}) do gaia.flags.set(f) end
      R.flagsPre = fr20.flagBytes(40)
      R.sceneBefore = kit.r16(0x0644)
      if P.scene then
        R.loadsBefore = loadStarts
        if P.dest then
          -- full warp: warpDestX/Y ($064C/$064E), warpFacing ($0650), warpScrollParams ($0652), then sceneNext last
          kit.w16(0x064C, P.dest[1]); kit.w16(0x064E, P.dest[2]); kit.w16(0x0650, P.dest[3]); kit.w16(0x0652, P.dest[4])
        end
        kit.w16(0x0642, P.scene)
        mm:go("loading")
      else
        R.arrivedF = kit.frame
        R.arrive36 = kit.r16(0x0036)
        mm:go("banner")
      end
    end,
    loading = function(mm)
      if loadStarts > R.loadsBefore and kit.r16(0x0642) == 0 and kit.r16(0x0644) == P.scene then
        R.arrive36 = kit.r16(0x0036)
        mm:go("await36")
      elseif mm:elapsed() > 900 then R.error = "load_timeout"; mm:go("done") end
    end,
    await36 = function(mm)
      -- the NMI (and so $36) runs again only when the scene setup has finished
      if kit.r16(0x0036) ~= R.arrive36 then
        R.arrivedF = kit.frame
        R.flags0 = fr20.flagBytes(40)
        mm:go("banner")
      elseif mm:elapsed() > 400 then
        R.arrivedF = kit.frame; R.no36 = true; R.flags0 = fr20.flagBytes(40)
        mm:go("banner")
      end
    end,
    banner = function(mm)
      if not R.flags0 then R.flags0 = fr20.flagBytes(40) end
      R.banner = R.banner or 0
      if kit.r8(0x00B4) ~= 0 then
        if not pulsing then kit.pulse({ "a" }, 3, 3); pulsing = true; R.bannerSeen = true end
        R.banner = R.banner + 1
      elseif pulsing then
        kit.stopPulse(); pulsing = false
        mm:go("watch0")
      elseif mm:elapsed() > 20 and not R.bannerSeen then
        mm:go("watch0")
      end
      if mm:elapsed() > 600 then R.bannerStuck = true; kit.stopPulse(); pulsing = false; mm:go("watch0") end
    end,
    watch0 = function(mm)
      R.mask0, R.pf0, R.dmf0 = kit.r16(0x065A), kit.r16(0x09AE), kit.r16(0x09EC)
      R.pos0 = { kit.r16(0x1014), kit.r16(0x1016) }
      R.sc0 = kit.r16(0x0644)
      R.actors0, R.act0 = actorsCompact()
      R.cells0 = cells3()
      R.n36start = kit.r16(0x0036)
      R.frame0 = kit.frame
      tl(0)
      mm:go("watch")
    end,
    watch = function(mm)
      n = n + 1
      tl(n)
      -- free = joypad mask clear, no script lock bit in playerFlags, no dialogue mode (displayModeFlags $2000), still in the target scene
      local free = kit.r16(0x065A) == 0 and (kit.r16(0x09AE) & 0x0800) == 0 and (kit.r16(0x09EC) & 0x2000) == 0
        and (not P.scene or kit.r16(0x0644) == P.scene)
      if free then
        freeRun = freeRun + 1
        if freeRun == 20 and not R.releaseN then R.releaseN = n - 19 end
      else freeRun = 0 end
      R.n = n
      if n >= watch or (P.stopFree and freeRun >= P.stopFree) then mm:go("done") end
    end,
    done = function(mm)
      kit.stopPulse()
      R.flags1 = fr20.flagBytes(40)
      R.mask1, R.pf1, R.dmf1 = kit.r16(0x065A), kit.r16(0x09AE), kit.r16(0x09EC)
      R.pos1 = { kit.r16(0x1014), kit.r16(0x1016) }
      R.actors1, R.act1 = actorsCompact()
      R.cnt0DBC = kit.r16(0x0DBC)
      R.hp1, R.hpMax1 = kit.r16(0x0ACE), kit.r16(0x0ACA)
      R.sceneEnd = kit.r16(0x0644)
      R.bgm = kit.r16(0x06F2)
      R.bounds = { kit.r16(0x0692), kit.r16(0x0696), kit.r8(0x0693) }
      R.cells1 = cells3()
      R.d36 = R.n36start and ((kit.r16(0x0036) - R.n36start) & 0xFFFF) or nil
      R.frames = kit.frame
      R.k, R.sp = cpuSane()
      R.p1010 = kit.r16(0x1010)
      kit.finish(R)
    end,
  } }
  if (P.input or "none") == "mashA" then
    emu.addEventCallback(function()
      if m.state == "watch" and (n % 12) < 2 then emu.setInput({ a = true }, 0) end
    end, emu.eventType.inputPolled)
  end
  return m, R
end

function fr40.move(P)
  P = P or {}
  local dirs = P.dirs or { "down", "up", "left", "right" }
  local hold, gap = P.hold or 20, P.gap or 8
  local R = { seg = {}, ctl36 = 0 }
  emu.addMemoryCallback(function(a, v) if (a & 0xFFFF) == 0x36 then R.ctl36 = R.ctl36 + 1 end end,
    emu.callbackType.write, 0x36, 0x36, emu.cpuType.snes, gaia.WR)
  R.sc0, R.mask0, R.pf0, R.dmf0 = kit.r16(0x0644), kit.r16(0x065A), kit.r16(0x09AE), kit.r16(0x09EC)
  R.pos0 = { kit.r16(0x1014), kit.r16(0x1016) }
  local c = 0
  local prev = { R.pos0[1], R.pos0[2] }
  local total = #dirs * (hold + gap)
  local cur
  emu.addEventCallback(function()
    local f = c + 1
    for i, d in ipairs(dirs) do
      local s = (i - 1) * (hold + gap) + 1
      if f >= s and f < s + hold then emu.setInput({ [d] = true }, 0) end
    end
  end, emu.eventType.inputPolled)
  kit.onFrame(function()
    c = c + 1
    local pos = { kit.r16(0x1014), kit.r16(0x1016) }
    for i, d in ipairs(dirs) do
      local s = (i - 1) * (hold + gap) + 1
      if c == s then R.seg[i] = { dir = d, x0 = prev[1], y0 = prev[2] } end
      if c == s + hold - 1 then
        local g = R.seg[i]
        g.x1, g.y1 = pos[1], pos[2]
        g.dx = ((pos[1] - g.x0 + 0x8000) & 0xFFFF) - 0x8000
        g.dy = ((pos[2] - g.y0 + 0x8000) & 0xFFFF) - 0x8000
        g.sc = kit.r16(0x0644)
        g.mask = kit.r16(0x065A)
        g.dmf = kit.r16(0x09EC)
        if g.sc ~= R.sc0 and not R.exitAt then R.exitAt = i end
      end
    end
    prev = pos
    if c >= total then
      R.mask1, R.pf1, R.sc1, R.dmf1 = kit.r16(0x065A), kit.r16(0x09AE), kit.r16(0x0644), kit.r16(0x09EC)
      R.pos1 = pos
      R.frames = c
      kit.finish(R)
    end
  end)
end
```
